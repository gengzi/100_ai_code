### 题目1：大规模边缘AI推理与联邦学习架构 ⭐⭐⭐⭐⭐
**问题**：设计一个支持百万边缘设备的AI推理系统，实现边缘推理、模型更新、联邦学习和全局聚合的完整架构。

**核心架构组件**：
1. **边缘AI推理引擎** - 轻量级模型部署、实时推理、资源管理
2. **联邦学习协调器** - 模型分发、梯度聚合、安全机制
3. **边缘设备管理** - 设备注册、健康监控、资源调度
4. **模型优化器** - 模型压缩、量化、知识蒸馏

import java.util.*;
import java.util.concurrent.*;
import java.util.stream.*;
import java.time.*;
import java.security.*;

/**
 * 边缘AI推理与联邦学习系统
 */
public class EdgeAISystem {

    // 系统主控制器
    public static class EdgeAIController {
        private final EdgeDeviceManager deviceManager;
        private final FederatedLearningCoordinator flCoordinator;
        private final ModelOptimizer modelOptimizer;
        private final InferenceEngine inferenceEngine;
        private final SecurityManager securityManager;
        private final MonitoringService monitoringService;
        private final ExecutorService executorService;

        public EdgeAIController() {
            this.deviceManager = new EdgeDeviceManager();
            this.flCoordinator = new FederatedLearningCoordinator();
            this.modelOptimizer = new ModelOptimizer();
            this.inferenceEngine = new InferenceEngine();
            this.securityManager = new SecurityManager();
            this.monitoringService = new MonitoringService();
            this.executorService = Executors.newFixedThreadPool(50);

            initializeSystem();
        }

        private void initializeSystem() {
            // 初始化安全组件
            securityManager.initializeEncryption();

            // 启动监控系统
            monitoringService.startMonitoring();

            // 启动设备发现
            deviceManager.startDeviceDiscovery();

            System.out.println("边缘AI系统初始化完成");
        }

        // 注册边缘设备
        public CompletableFuture<DeviceRegistrationResult> registerDevice(EdgeDevice device) {
            return CompletableFuture.supplyAsync(() -> {
                try {
                    // 安全验证
                    if (!securityManager.validateDevice(device)) {
                        return new DeviceRegistrationResult(device.getId(), false, "设备验证失败");
                    }

                    // 设备注册
                    boolean success = deviceManager.registerDevice(device);

                    if (success) {
                        // 部署初始模型
                        deployInitialModel(device);

                        // 启动监控
                        monitoringService.monitorDevice(device);

                        return new DeviceRegistrationResult(device.getId(), true, "注册成功");
                    } else {
                        return new DeviceRegistrationResult(device.getId(), false, "注册失败");
                    }

                } catch (Exception e) {
                    return new DeviceRegistrationResult(device.getId(), false, "异常: " + e.getMessage());
                }
            }, executorService);
        }

        // 推理请求处理
        public CompletableFuture<InferenceResult> performInference(String deviceId, InferenceRequest request) {
            return CompletableFuture.supplyAsync(() -> {
                try {
                    EdgeDevice device = deviceManager.getDevice(deviceId);
                    if (device == null) {
                        throw new RuntimeException("设备不存在: " + deviceId);
                    }

                    // 检查设备状态
                    if (!device.isHealthy()) {
                        throw new RuntimeException("设备不健康");
                    }

                    // 执行推理
                    long startTime = System.currentTimeMillis();
                    InferenceResult result = inferenceEngine.infer(device, request);
                    long latency = System.currentTimeMillis() - startTime;

                    // 更新统计信息
                    device.updateInferenceStats(latency, result.isSuccess());

                    // 记录监控数据
                    monitoringService.recordInference(deviceId, request, result, latency);

                    return result;

                } catch (Exception e) {
                    monitoringService.recordError(deviceId, "inference", e);
                    return new InferenceResult(false, null, "推理失败: " + e.getMessage());
                }
            }, executorService);
        }

        // 联邦学习：训练轮次
        public CompletableFuture<FederatedRoundResult> performFederatedRound(String globalModelId) {
            return CompletableFuture.supplyAsync(() -> {
                try {
                    long startTime = System.currentTimeMillis();

                    // 1. 选择参与设备
                    List<EdgeDevice> selectedDevices = selectDevicesForFL();

                    // 2. 分发全局模型
                    ModelDistributionResult distributionResult = distributeGlobalModel(globalModelId, selectedDevices);

                    // 3. 等待本地训练完成
                    List<LocalTrainingResult> localResults = waitForLocalTraining(selectedDevices);

                    // 4. 聚合本地模型
                    ModelAggregationResult aggregationResult = flCoordinator.aggregateModels(localResults);

                    // 5. 验证全局模型
                    ModelValidationResult validation = validateGlobalModel(aggregationResult.getAggregatedModel());

                    // 6. 更新全局模型
                    if (validation.isValid()) {
                        updateGlobalModel(globalModelId, aggregationResult.getAggregatedModel());
                    }

                    long processingTime = System.currentTimeMillis() - startTime;

                    return new FederatedRoundResult(
                        globalModelId,
                        selectedDevices.size(),
                        localResults.size(),
                        aggregationResult,
                        validation,
                        processingTime,
                        Instant.now()
                    );

                } catch (Exception e) {
                    return new FederatedRoundResult(globalModelId, 0, 0, null, null, 0, Instant.now());
                }
            }, executorService);
        }

        private void deployInitialModel(EdgeDevice device) {
            try {
                // 获取初始模型
                AIModel initialModel = modelOptimizer.getInitialModel(device.getType());

                // 模型优化（压缩、量化）
                AIModel optimizedModel = modelOptimizer.optimizeModel(initialModel, device);

                // 部署到设备
                deviceManager.deployModel(device.getId(), optimizedModel);

                System.out.printf("初始模型已部署到设备: %s%n", device.getId());

            } catch (Exception e) {
                System.err.printf("初始模型部署失败 %s: %s%n", device.getId(), e.getMessage());
            }
        }

        private List<EdgeDevice> selectDevicesForFL() {
            List<EdgeDevice> allDevices = deviceManager.getHealthyDevices();

            // 选择条件：设备健康、网络稳定、计算资源充足
            return allDevices.stream()
                .filter(device -> device.getResource().getAvailableMemory() > 512 * 1024 * 1024) // 512MB
                .filter(device -> device.getNetwork().getLatency() < 100) // 100ms
                .filter(device -> device.getResource().getCpuUsage() < 0.7) // CPU使用率<70%
                .limit(100) // 限制参与设备数量
                .collect(Collectors.toList());
        }

        private ModelDistributionResult distributeGlobalModel(String globalModelId, List<EdgeDevice> devices) {
            AIModel globalModel = flCoordinator.getGlobalModel(globalModelId);
            int successCount = 0;

            for (EdgeDevice device : devices) {
                try {
                    // 模型安全传输
                    EncryptedModel encryptedModel = securityManager.encryptModel(globalModel);

                    // 部署到设备
                    deviceManager.deployEncryptedModel(device.getId(), encryptedModel);
                    successCount++;

                } catch (Exception e) {
                    System.err.printf("模型分发失败 %s: %s%n", device.getId(), e.getMessage());
                }
            }

            return new ModelDistributionResult(devices.size(), successCount);
        }

        private List<LocalTrainingResult> waitForLocalTraining(List<EdgeDevice> devices) {
            List<LocalTrainingResult> results = new ArrayList<>();
            CountDownLatch latch = new CountDownLatch(devices.size());

            for (EdgeDevice device : devices) {
                CompletableFuture.supplyAsync(() -> {
                    try {
                        // 启动本地训练
                        LocalTrainingResult result = deviceManager.startLocalTraining(device);

                        synchronized (results) {
                            results.add(result);
                        }

                        return result;

                    } catch (Exception e) {
                        return new LocalTrainingResult(device.getId(), false, null, e.getMessage());
                    } finally {
                        latch.countDown();
                    }
                }, executorService);
            }

            // 等待所有设备完成（带超时）
            try {
                boolean completed = latch.await(30, TimeUnit.MINUTES);
                if (!completed) {
                    System.out.println("部分设备训练超时");
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }

            return results;
        }

        private ModelValidationResult validateGlobalModel(AIModel aggregatedModel) {
            // 验证聚合后的全局模型
            boolean isValid = modelOptimizer.validateModel(aggregatedModel);
            double accuracy = isValid ? 0.85 + Math.random() * 0.1 : 0.0; // 模拟准确率

            return new ModelValidationResult(isValid, accuracy, "模型验证完成");
        }

        private void updateGlobalModel(String globalModelId, AIModel aggregatedModel) {
            flCoordinator.updateGlobalModel(globalModelId, aggregatedModel);
            System.out.printf("全局模型已更新: %s%n", globalModelId);
        }

        // 获取系统状态
        public SystemStatus getSystemStatus() {
            return new SystemStatus(
                deviceManager.getDeviceCount(),
                deviceManager.getHealthyDeviceCount(),
                flCoordinator.getGlobalModelCount(),
                inferenceEngine.getTotalInferences(),
                monitoringService.getAverageLatency()
            );
        }
    }

    // 边缘设备管理器
    public static class EdgeDeviceManager {
        private final Map<String, EdgeDevice> devices = new ConcurrentHashMap<>();
        private final ScheduledExecutorService scheduler = Executors.newScheduledThreadPool(5);

        public void startDeviceDiscovery() {
            // 定期检查设备健康状态
            scheduler.scheduleAtFixedRate(this::checkDeviceHealth, 0, 30, TimeUnit.SECONDS);

            // 定期清理离线设备
            scheduler.scheduleAtFixedRate(this::cleanupOfflineDevices, 0, 5, TimeUnit.MINUTES);
        }

        public boolean registerDevice(EdgeDevice device) {
            if (devices.containsKey(device.getId())) {
                return false; // 设备已存在
            }

            devices.put(device.getId(), device);
            device.setStatus(DeviceStatus.ONLINE);
            device.setLastHeartbeat(Instant.now());

            return true;
        }

        public void deployModel(String deviceId, AIModel model) {
            EdgeDevice device = devices.get(deviceId);
            if (device != null) {
                device.deployModel(model);
            }
        }

        public void deployEncryptedModel(String deviceId, EncryptedModel encryptedModel) {
            EdgeDevice device = devices.get(deviceId);
            if (device != null) {
                device.deployEncryptedModel(encryptedModel);
            }
        }

        public LocalTrainingResult startLocalTraining(EdgeDevice device) {
            return device.startLocalTraining();
        }

        public EdgeDevice getDevice(String deviceId) {
            return devices.get(deviceId);
        }

        public List<EdgeDevice> getHealthyDevices() {
            return devices.values().stream()
                .filter(EdgeDevice::isHealthy)
                .collect(Collectors.toList());
        }

        public int getDeviceCount() {
            return devices.size();
        }

        public int getHealthyDeviceCount() {
            return (int) devices.values().stream().filter(EdgeDevice::isHealthy).count();
        }

        private void checkDeviceHealth() {
            Instant now = Instant.now();

            devices.values().forEach(device -> {
                if (Duration.between(device.getLastHeartbeat(), now).toSeconds() > 120) {
                    device.setStatus(DeviceStatus.OFFLINE);
                }
            });
        }

        private void cleanupOfflineDevices() {
            devices.entrySet().removeIf(entry -> {
                EdgeDevice device = entry.getValue();
                return device.getStatus() == DeviceStatus.OFFLINE &&
                       Duration.between(device.getLastHeartbeat(), Instant.now()).toMinutes() > 30;
            });
        }
    }

    // 联邦学习协调器
    public static class FederatedLearningCoordinator {
        private final Map<String, AIModel> globalModels = new ConcurrentHashMap<>();
        private final AggregationStrategy aggregationStrategy = new FedAvgStrategy();
        private final ExecutorService aggregationExecutor = Executors.newFixedThreadPool(10);

        public AIModel getGlobalModel(String modelId) {
            return globalModels.get(modelId);
        }

        public void updateGlobalModel(String modelId, AIModel model) {
            globalModels.put(modelId, model);
        }

        public int getGlobalModelCount() {
            return globalModels.size();
        }

        public ModelAggregationResult aggregateModels(List<LocalTrainingResult> localResults) {
            return CompletableFuture.supplyAsync(() -> {
                try {
                    // 过滤成功的本地训练结果
                    List<AIModel> validModels = localResults.stream()
                        .filter(LocalTrainingResult::isSuccess)
                        .map(LocalTrainingResult::getModel)
                        .filter(Objects::nonNull)
                        .collect(Collectors.toList());

                    if (validModels.isEmpty()) {
                        throw new RuntimeException("没有有效的本地模型");
                    }

                    // 聚合模型
                    AIModel aggregatedModel = aggregationStrategy.aggregate(validModels);

                    // 计算聚合统计
                    AggregationStats stats = calculateAggregationStats(validModels);

                    return new ModelAggregationResult(aggregatedModel, stats, localResults.size(), validModels.size());

                } catch (Exception e) {
                    throw new RuntimeException("模型聚合失败", e);
                }
            }, aggregationExecutor).join();
        }

        private AggregationStats calculateAggregationStats(List<AIModel> models) {
            double avgModelSize = models.stream()
                .mapToDouble(AIModel::getSize)
                .average().orElse(0);

            double avgTrainingTime = models.stream()
                .mapToDouble(AIModel::getTrainingTime)
                .average().orElse(0);

            return new AggregationStats(models.size(), avgModelSize, avgTrainingTime);
        }

        // FedAvg聚合策略
        public static class FedAvgStrategy implements AggregationStrategy {
            @Override
            public AIModel aggregate(List<AIModel> models) {
                // 简化的FedAvg实现
                AIModel aggregatedModel = new AIModel("aggregated_model");

                // 聚合权重
                double[] weights = new double[100]; // 假设100个参数
                Arrays.fill(weights, 0.0);

                for (AIModel model : models) {
                    double[] modelWeights = model.getWeights();
                    for (int i = 0; i < Math.min(weights.length, modelWeights.length); i++) {
                        weights[i] += modelWeights[i];
                    }
                }

                // 平均权重
                for (int i = 0; i < weights.length; i++) {
                    weights[i] /= models.size();
                }

                aggregatedModel.setWeights(weights);
                return aggregatedModel;
            }
        }

        // 聚合策略接口
        public interface AggregationStrategy {
            AIModel aggregate(List<AIModel> models);
        }
    }

    // 模型优化器
    public static class ModelOptimizer {
        private final Map<DeviceType, AIModel> initialModels = new HashMap<>();

        public ModelOptimizer() {
            initializeInitialModels();
        }

        private void initializeInitialModels() {
            // 初始化不同设备类型的初始模型
            initialModels.put(DeviceType.MOBILE, createMobileModel());
            initialModels.put(DeviceType.IOT, createIoTModel());
            initialModels.put(DeviceType.EDGE_SERVER, createEdgeServerModel());
        }

        public AIModel getInitialModel(DeviceType deviceType) {
            return initialModels.get(deviceType);
        }

        public AIModel optimizeModel(AIModel model, EdgeDevice device) {
            // 根据设备能力优化模型
            AIModel optimizedModel = model.clone();

            // 模型压缩
            if (device.getResource().getAvailableMemory() < 256 * 1024 * 1024) { // 256MB
                optimizedModel = compressModel(optimizedModel, 0.5);
            }

            // 量化
            optimizedModel = quantizeModel(optimizedModel, QuantizationType.INT8);

            return optimizedModel;
        }

        public boolean validateModel(AIModel model) {
            // 验证模型结构和参数
            return model != null && model.getWeights() != null && model.getWeights().length > 0;
        }

        private AIModel createMobileModel() {
            AIModel model = new AIModel("mobile_initial_model");
            double[] weights = new double[50]; // 移动设备使用更小的模型
            Arrays.fill(weights, Math.random() * 0.1);
            model.setWeights(weights);
            return model;
        }

        private AIModel createIoTModel() {
            AIModel model = new AIModel("iot_initial_model");
            double[] weights = new double[20]; // IoT设备使用最小模型
            Arrays.fill(weights, Math.random() * 0.1);
            model.setWeights(weights);
            return model;
        }

        private AIModel createEdgeServerModel() {
            AIModel model = new AIModel("edge_server_initial_model");
            double[] weights = new double[200]; // 边缘服务器使用较大模型
            Arrays.fill(weights, Math.random() * 0.1);
            model.setWeights(weights);
            return model;
        }

        private AIModel compressModel(AIModel model, double compressionRatio) {
            AIModel compressed = model.clone();
            double[] originalWeights = model.getWeights();
            int newSize = (int) (originalWeights.length * compressionRatio);
            double[] compressedWeights = Arrays.copyOf(originalWeights, newSize);
            compressed.setWeights(compressedWeights);
            return compressed;
        }

        private AIModel quantizeModel(AIModel model, QuantizationType quantizationType) {
            AIModel quantized = model.clone();
            double[] weights = model.getWeights();

            // 简化量化实现
            for (int i = 0; i < weights.length; i++) {
                switch (quantizationType) {
                    case INT8:
                        weights[i] = (byte) (weights[i] * 127);
                        break;
                    case INT16:
                        weights[i] = (short) (weights[i] * 32767);
                        break;
                }
            }

            quantized.setWeights(weights);
            return quantized;
        }
    }

    // 推理引擎
    public static class InferenceEngine {
        private final AtomicLong totalInferences = new AtomicLong(0);

        public InferenceResult infer(EdgeDevice device, InferenceRequest request) {
            long inferenceId = totalInferences.incrementAndGet();

            try {
                // 获取设备当前模型
                AIModel model = device.getCurrentModel();
                if (model == null) {
                    throw new RuntimeException("设备上没有可用的模型");
                }

                // 执行推理
                double[] input = request.getInput();
                double[] output = performInference(model, input);

                return new InferenceResult(true, output, "推理成功");

            } catch (Exception e) {
                return new InferenceResult(false, null, "推理失败: " + e.getMessage());
            }
        }

        private double[] performInference(AIModel model, double[] input) {
            // 简化的推理实现
            double[] weights = model.getWeights();
            double[] output = new double[10]; // 假设10个输出类别

            for (int i = 0; i < output.length; i++) {
                double sum = 0;
                for (int j = 0; j < Math.min(input.length, weights.length); j++) {
                    sum += input[j] * weights[j % weights.length];
                }
                output[i] = Math.tanh(sum); // tanh激活
            }

            // Softmax
            double sumExp = Arrays.stream(output).map(Math::exp).sum();
            for (int i = 0; i < output.length; i++) {
                output[i] = Math.exp(output[i]) / sumExp;
            }

            return output;
        }

        public long getTotalInferences() {
            return totalInferences.get();
        }
    }

    // 监控服务
    public static class MonitoringService {
        private final Map<String, DeviceMetrics> deviceMetrics = new ConcurrentHashMap<>();
        private final Queue<InferenceMetrics> inferenceMetrics = new ConcurrentLinkedQueue<>();
        private final AtomicLong totalLatency = new AtomicLong(0);
        private final AtomicLong inferenceCount = new AtomicLong(0);

        public void startMonitoring() {
            System.out.println("监控服务启动");
        }

        public void monitorDevice(EdgeDevice device) {
            deviceMetrics.put(device.getId(), new DeviceMetrics(device.getId()));
        }

        public void recordInference(String deviceId, InferenceRequest request, InferenceResult result, long latency) {
            inferenceCount.incrementAndGet();
            totalLatency.addAndGet(latency);

            InferenceMetrics metrics = new InferenceMetrics(deviceId, request, result, latency);
            inferenceMetrics.offer(metrics);

            // 更新设备指标
            DeviceMetrics deviceMetrics = this.deviceMetrics.get(deviceId);
            if (deviceMetrics != null) {
                deviceMetrics.addInference(latency, result.isSuccess());
            }
        }

        public void recordError(String deviceId, String operation, Exception e) {
            System.err.printf("设备错误 %s [%s]: %s%n", deviceId, operation, e.getMessage());
        }

        public double getAverageLatency() {
            long count = inferenceCount.get();
            return count > 0 ? (double) totalLatency.get() / count : 0;
        }

        public SystemMetrics getSystemMetrics() {
            return new SystemMetrics(
                deviceMetrics.size(),
                inferenceCount.get(),
                getAverageLatency(),
                calculateSuccessRate()
            );
        }

        private double calculateSuccessRate() {
            // 简化实现
            return 0.95; // 假设95%成功率
        }
    }

    public enum DeviceStatus { ONLINE, OFFLINE, TRAINING }
    public enum QuantizationType { INT8, INT16, FLOAT16 }

    public static class EdgeDevice {
        private final String id;
        private final String deviceId;
        private final DeviceType type;
        private final DeviceResource resource;
        private final NetworkInfo network;
        private volatile DeviceStatus status;
        private volatile Instant lastHeartbeat;
        private AIModel currentModel;
        private InferenceStats inferenceStats;

        public EdgeDevice(String id, String deviceId, DeviceType type, DeviceResource resource, NetworkInfo network) {
            this.id = id;
            this.deviceId = deviceId;
            this.type = type;
            this.resource = resource;
            this.network = network;
            this.status = DeviceStatus.OFFLINE;
            this.inferenceStats = new InferenceStats();
        }

        public void deployModel(AIModel model) {
            this.currentModel = model;
        }

        public void deployEncryptedModel(EncryptedModel encryptedModel) {
            // 解密并部署模型
            this.currentModel = new SecurityManager().decryptModel(encryptedModel);
        }

        public LocalTrainingResult startLocalTraining() {
            if (currentModel == null) {
                return new LocalTrainingResult(id, false, null, "没有可用模型");
            }

            try {
                // 模拟本地训练过程
                Thread.sleep((long) (5000 + Math.random() * 10000)); // 5-15秒

                // 创建本地更新的模型
                AIModel localModel = currentModel.clone();
                double[] weights = localModel.getWeights();
                for (int i = 0; i < weights.length; i++) {
                    weights[i] += (Math.random() - 0.5) * 0.01; // 小幅更新
                }
                localModel.setWeights(weights);
                localModel.setTrainingTime(5000 + Math.random() * 10000);

                return new LocalTrainingResult(id, true, localModel, "训练成功");

            } catch (Exception e) {
                return new LocalTrainingResult(id, false, null, "训练失败: " + e.getMessage());
            }
        }

        public boolean isHealthy() {
            return status == DeviceStatus.ONLINE && Duration.between(lastHeartbeat, Instant.now()).toSeconds() < 120;
        }

        public void updateInferenceStats(long latency, boolean success) {
            inferenceStats.update(latency, success);
        }

        // Getters
        public String getId() { return id; }
        public String getDeviceId() { return deviceId; }
        public DeviceType getType() { return type; }
        public DeviceResource getResource() { return resource; }
        public NetworkInfo getNetwork() { return network; }
        public DeviceStatus getStatus() { return status; }
        public Instant getLastHeartbeat() { return lastHeartbeat; }
        public AIModel getCurrentModel() { return currentModel; }
        public InferenceStats getInferenceStats() { return inferenceStats; }

        public void setStatus(DeviceStatus status) { this.status = status; }
        public void setLastHeartbeat(Instant lastHeartbeat) { this.lastHeartbeat = lastHeartbeat; }
    }

    public static class DeviceResource {
        private final long totalMemory;
        private final double cpuCores;
        private volatile double availableMemory;
        private volatile double cpuUsage;

        public DeviceResource(long totalMemory, double cpuCores) {

    public static class InferenceStats {
        private final AtomicLong totalInferences = new AtomicLong(0);
        private final AtomicLong successfulInferences = new AtomicLong(0);
        private final AtomicLong totalLatency = new AtomicLong(0);

        public void update(long latency, boolean success) {
            totalInferences.incrementAndGet();
            if (success) {
                successfulInferences.incrementAndGet();
            }
            totalLatency.addAndGet(latency);
        }

        public long getTotalInferences() { return totalInferences.get(); }
        public long getSuccessfulInferences() { return successfulInferences.get(); }
        public double getSuccessRate() {
            long total = totalInferences.get();
            return total > 0 ? (double) successfulInferences.get() / total : 0;
        }
        public double getAverageLatency() {
            long total = totalInferences.get();
            return total > 0 ? (double) totalLatency.get() / total : 0;
        }
    }

        EdgeAIController controller = new EdgeAIController();

            registrationFutures.add(future);
        }

        // 等待设备注册完成
        for (CompletableFuture<DeviceRegistrationResult> future : registrationFutures) {
            DeviceRegistrationResult result = future.get(10, TimeUnit.SECONDS);

        }

        // 模拟推理请求

            InferenceRequest request = new InferenceRequest("req_" + i, input);
            CompletableFuture<InferenceResult> future = controller.performInference(deviceId, request);
            inferenceFutures.add(future);
        }

        // 等待推理完成
        for (int i = 0; i < inferenceFutures.size(); i++) {
            InferenceResult result = inferenceFutures.get(i).get(30, TimeUnit.SECONDS);
            if (result.isSuccess()) {
                System.out.printf("推理 %d 成功，输出前3个值: [%.3f, %.3f, %.3f]%n",
                                 i + 1,
                                 result.getOutput()[0], result.getOutput()[1], result.getOutput()[2]);
            } else {
                System.out.printf("推理 %d 失败: %s%n", i + 1, result.getMessage());
            }
        }

        // 模拟联邦学习轮次
        System.out.println("\n开始联邦学习轮次...");
        CompletableFuture<FederatedRoundResult> flFuture = controller.performFederatedRound("global_model_1");
        FederatedRoundResult flResult = flFuture.get(10, TimeUnit.MINUTES);

        System.out.println("=== 联邦学习结果 ===");
        System.out.printf("全局模型ID: %s%n", flResult.getGlobalModelId());
        System.out.printf("选择设备数: %d%n", flResult.getSelectedDevices());
        System.out.printf("完成设备数: %d%n", flResult.getCompletedDevices());
        System.out.printf("处理时间: %d ms%n", flResult.getProcessingTime());

        if (flResult.getAggregationResult() != null) {
            AggregationStats stats = flResult.getAggregationResult().getStats();
            System.out.printf("聚合统计: 参与者=%d, 平均模型大小=%.2fKB, 平均训练时间=%.2fs%n",
                             stats.getParticipantCount(), stats.getAverageModelSize() / 1024, stats.getAverageTrainingTime() / 1000);
        }

        // 获取系统状态
        SystemStatus status = controller.getSystemStatus();
        System.out.println("\n=== 系统状态 ===");
        System.out.printf("总设备数: %d%n", status.getTotalDevices());
        System.out.printf("健康设备数: %d%n", status.getHealthyDevices());
        System.out.printf("全局模型数: %d%n", status.getGlobalModels());
        System.out.printf("总推理数: %d%n", status.getTotalInferences());
        System.out.printf("平均延迟: %.2f ms%n", status.getAverageLatency());

        Thread.sleep(2000);
        controller.shutdown();
        System.out.println("边缘AI系统已关闭");
    }
