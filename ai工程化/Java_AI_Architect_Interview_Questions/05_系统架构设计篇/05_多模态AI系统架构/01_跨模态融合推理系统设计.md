# 跨模态融合推理系统设计

## 🎯 面试题目

### 题目1：多模态AI融合架构设计 ⭐⭐⭐⭐⭐
**问题**：设计一个支持文本、图像、音频多模态融合的AI系统，包括特征提取、跨模态对齐、融合推理和输出生成的完整架构。

**核心架构组件**：
1. **单模态编码器**：文本编码器、图像编码器、音频编码器
2. **跨模态对齐**：注意力机制、对比学习、多模态嵌入空间
3. **融合推理**：早期融合、晚期融合、混合融合策略
4. **输出生成**：多任务学习、条件生成、跨模态翻译

**核心代码示例**：

```java
import java.util.*;
import java.util.concurrent.*;

/**
 * 多模态AI融合系统架构
 */
public class MultiModalAISystem {

    // 多模态处理器管理器
    public static class MultiModalProcessorManager {
        private final Map<ModalityType, ModalityProcessor> processors;
        private final CrossModalFusion fusionEngine;
        private final OutputGenerator outputGenerator;
        private final ExecutorService executorService;

        public MultiModalProcessorManager() {
            this.processors = new ConcurrentHashMap<>();
            this.fusionEngine = new CrossModalFusion();
            this.outputGenerator = new OutputGenerator();
            this.executorService = Executors.newFixedThreadPool(8);

            // 初始化模态处理器
            initializeProcessors();
        }

        private void initializeProcessors() {
            processors.put(ModalityType.TEXT, new TextProcessor());
            processors.put(ModalityType.IMAGE, new ImageProcessor());
            processors.put(ModalityType.AUDIO, new AudioProcessor());
            processors.put(ModalityType.VIDEO, new VideoProcessor());
        }

        public CompletableFuture<MultiModalResult> process(MultiModalInput input) {
            return CompletableFuture.supplyAsync(() -> {
                try {
                    // 1. 并行处理各模态数据
                    Map<ModalityType, Future<ModalityFeatures>> futures = new HashMap<>();

                    for (Map.Entry<ModalityType, Object> entry : input.getData().entrySet()) {
                        ModalityType modality = entry.getKey();
                        Object data = entry.getValue();

                        Future<ModalityFeatures> future = executorService.submit(() -> {
                            ModalityProcessor processor = processors.get(modality);
                            return processor.extractFeatures(data);
                        });

                        futures.put(modality, future);
                    }

                    // 2. 等待所有模态处理完成
                    Map<ModalityType, ModalityFeatures> features = new HashMap<>();
                    for (Map.Entry<ModalityType, Future<ModalityFeatures>> entry : futures.entrySet()) {
                        features.put(entry.getKey(), entry.getValue().get(30, TimeUnit.SECONDS));
                    }

                    // 3. 跨模态融合
                    FusedFeatures fusedFeatures = fusionEngine.fuse(features, input.getTask());

                    // 4. 生成输出
                    Object output = outputGenerator.generate(fusedFeatures, input.getTask());

                    return new MultiModalResult(fusedFeatures, output, System.currentTimeMillis());

                } catch (Exception e) {
                    throw new RuntimeException("多模态处理失败", e);
                }
            }, executorService);
        }
    }

    // 跨模态融合引擎
    public static class CrossModalFusion {
        private final AttentionMechanism attentionMechanism;
        private final FeatureAlignment featureAlignment;
        private final FusionStrategy fusionStrategy;

        public CrossModalFusion() {
            this.attentionMechanism = new CrossModalAttention();
            this.featureAlignment = new FeatureAlignment();
            this.fusionStrategy = new AdaptiveFusionStrategy();
        }

        public FusedFeatures fuse(Map<ModalityType, ModalityFeatures> features, TaskType task) {
            // 1. 特征对齐
            Map<ModalityType, float[]> alignedFeatures = featureAlignment.alignFeatures(features);

            // 2. 计算跨模态注意力
            Map<ModalityType, float[]> attentionWeights = attentionMechanism.computeAttention(alignedFeatures);

            // 3. 融合策略选择和应用
            float[] fusedVector = fusionStrategy.fuse(alignedFeatures, attentionWeights, task);

            return new FusedFeatures(fusedVector, alignedFeatures, attentionWeights);
        }
    }

    // 跨模态注意力机制
    public static class CrossModalAttention {
        private final int featureDim;
        private final float[][] projectionMatrices;

        public CrossModalAttention() {
            this.featureDim = 512;
            this.projectionMatrices = new float[4][featureDim * featureDim]; // 文本、图像、音频、视频
            initializeProjections();
        }

        private void initializeProjections() {
            Random random = new Random(42);
            for (int i = 0; i < projectionMatrices.length; i++) {
                for (int j = 0; j < projectionMatrices[i].length; j++) {
                    projectionMatrices[i][j] = (float) (random.nextGaussian() * 0.02);
                }
            }
        }

        public Map<ModalityType, float[]> computeAttention(Map<ModalityType, float[]> features) {
            Map<ModalityType, float[]> attentionWeights = new HashMap<>();

            // 计算每个模态对其他模态的注意力
            for (Map.Entry<ModalityType, float[]> entry1 : features.entrySet()) {
                ModalityType sourceModality = entry1.getKey();
                float[] sourceFeatures = entry1.getValue();

                float[] attentionVector = new float[features.size()];
                int index = 0;

                for (Map.Entry<ModalityType, float[]> entry2 : features.entrySet()) {
                    ModalityType targetModality = entry2.getKey();
                    float[] targetFeatures = entry2.getValue();

                    // 计算相似度/注意力权重
                    float similarity = computeSimilarity(sourceFeatures, targetFeatures);
                    attentionVector[index++] = similarity;
                }

                // Softmax归一化
                attentionVector = softmax(attentionVector);
                attentionWeights.put(sourceModality, attentionVector);
            }

            return attentionWeights;
        }

        private float computeSimilarity(float[] features1, float[] features2) {
            float dotProduct = 0;
            float norm1 = 0;
            float norm2 = 0;

            for (int i = 0; i < Math.min(features1.length, features2.length); i++) {
                dotProduct += features1[i] * features2[i];
                norm1 += features1[i] * features1[i];
                norm2 += features2[i] * features2[i];
            }

            if (norm1 == 0 || norm2 == 0) return 0;
            return dotProduct / ((float) Math.sqrt(norm1 * norm2));
        }

        private float[] softmax(float[] input) {
            float max = Float.NEGATIVE_INFINITY;
            for (float v : input) {
                max = Math.max(max, v);
            }

            float sum = 0;
            float[] output = new float[input.length];
            for (int i = 0; i < input.length; i++) {
                output[i] = (float) Math.exp(input[i] - max);
                sum += output[i];
            }

            for (int i = 0; i < output.length; i++) {
                output[i] /= sum;
            }

            return output;
        }
    }

    // 特征对齐器
    public static class FeatureAlignment {
        private final int targetDim = 512;

        public Map<ModalityType, float[]> alignFeatures(Map<ModalityType, ModalityFeatures> features) {
            Map<ModalityType, float[]> alignedFeatures = new HashMap<>();

            for (Map.Entry<ModalityType, ModalityFeatures> entry : features.entrySet()) {
                ModalityType modality = entry.getKey();
                ModalityFeatures featuresObj = entry.getValue();

                float[] originalFeatures = featuresObj.getFeatures();
                float[] aligned = alignToTargetDimension(originalFeatures, modality);

                alignedFeatures.put(modality, aligned);
            }

            return alignedFeatures;
        }

        private float[] alignToTargetDimension(float[] features, ModalityType modality) {
            // 根据模态类型使用不同的对齐策略
            switch (modality) {
                case TEXT:
                    return projectTextFeatures(features);
                case IMAGE:
                    return projectImageFeatures(features);
                case AUDIO:
                    return projectAudioFeatures(features);
                case VIDEO:
                    return projectVideoFeatures(features);
                default:
                    return resizeFeatures(features, targetDim);
            }
        }

        private float[] projectTextFeatures(float[] features) {
            // 文本特征投影
            return resizeFeatures(features, targetDim);
        }

        private float[] projectImageFeatures(float[] features) {
            // 图像特征投影
            return resizeFeatures(features, targetDim);
        }

        private float[] projectAudioFeatures(float[] features) {
            // 音频特征投影
            return resizeFeatures(features, targetDim);
        }

        private float[] projectVideoFeatures(float[] features) {
            // 视频特征投影
            return resizeFeatures(features, targetDim);
        }

        private float[] resizeFeatures(float[] original, int targetDim) {
            float[] resized = new float[targetDim];
            if (original.length == targetDim) {
                System.arraycopy(original, 0, resized, 0, targetDim);
            } else {
                // 简单的插值重采样
                for (int i = 0; i < targetDim; i++) {
                    int sourceIndex = (int) (i * (double) original.length / targetDim);
                    resized[i] = original[sourceIndex];
                }
            }
            return resized;
        }
    }

    // 自适应融合策略
    public static class AdaptiveFusionStrategy {
        public float[] fuse(Map<ModalityType, float[]> features,
                           Map<ModalityType, float[]> attentionWeights,
                           TaskType task) {
            switch (task) {
                case MULTIMODAL_CLASSIFICATION:
                    return weightedFusion(features, attentionWeights);
                case CROSS_MODAL_GENERATION:
                    return attentionFusion(features, attentionWeights);
                case MULTIMODAL_RETRIEVAL:
                    return concatenationFusion(features);
                default:
                    return weightedFusion(features, attentionWeights);
            }
        }

        private float[] weightedFusion(Map<ModalityType, float[]> features,
                                     Map<ModalityType, float[]> attentionWeights) {
            if (features.isEmpty()) return new float[0];

            int featureDim = features.values().iterator().next().length;
            float[] fused = new float[featureDim];

            for (Map.Entry<ModalityType, float[]> entry : features.entrySet()) {
                ModalityType modality = entry.getKey();
                float[] modalityFeatures = entry.getValue();
                float[] weights = attentionWeights.get(modality);

                // 计算该模态的整体权重
                float modalityWeight = 0;
                for (float weight : weights) {
                    modalityWeight += weight;
                }
                modalityWeight /= weights.length;

                // 加权累加
                for (int i = 0; i < featureDim; i++) {
                    fused[i] += modalityFeatures[i] * modalityWeight;
                }
            }

            return fused;
        }

        private float[] attentionFusion(Map<ModalityType, float[]> features,
                                       Map<ModalityType, float[]> attentionWeights) {
            // 基于注意力的融合策略
            return weightedFusion(features, attentionWeights);
        }

        private float[] concatenationFusion(Map<ModalityType, float[]> features) {
            // 简单的连接融合
            int totalDim = features.values().stream()
                .mapToInt(arr -> arr.length)
                .sum();

            float[] concatenated = new float[totalDim];
            int offset = 0;

            for (float[] modalityFeatures : features.values()) {
                System.arraycopy(modalityFeatures, 0, concatenated, offset, modalityFeatures.length);
                offset += modalityFeatures.length;
            }

            return concatenated;
        }
    }

    // 输出生成器
    public static class OutputGenerator {
        public Object generate(FusedFeatures fusedFeatures, TaskType task) {
            switch (task) {
                case MULTIMODAL_CLASSIFICATION:
                    return generateClassification(fusedFeatures);
                case CROSS_MODAL_GENERATION:
                    return generateCrossModalOutput(fusedFeatures);
                case MULTIMODAL_RETRIEVAL:
                    return generateRetrievalResult(fusedFeatures);
                default:
                    return generateDefaultOutput(fusedFeatures);
            }
        }

        private String generateClassification(FusedFeatures fusedFeatures) {
            // 模拟分类任务
            float[] features = fusedFeatures.getFusedVector();
            float confidence = Math.abs(features[0]) % 1.0f;
            String label = confidence > 0.5 ? "正面" : "负面";
            return String.format("分类结果: %s (置信度: %.2f)", label, confidence);
        }

        private String generateCrossModalOutput(FusedFeatures fusedFeatures) {
            // 模拟跨模态生成
            return "跨模态生成结果: 这是一个基于多模态输入生成的文本描述";
        }

        private List<String> generateRetrievalResult(FusedFeatures fusedFeatures) {
            // 模拟检索任务
            return Arrays.asList(
                "检索结果1: 相关性0.89",
                "检索结果2: 相关性0.76",
                "检索结果3: 相关性0.65"
            );
        }

        private String generateDefaultOutput(FusedFeatures fusedFeatures) {
            return "默认输出: " + Arrays.toString(Arrays.copyOf(fusedFeatures.getFusedVector(), 5));
        }
    }

    // 数据结构定义
    public enum ModalityType {
        TEXT, IMAGE, AUDIO, VIDEO
    }

    public enum TaskType {
        MULTIMODAL_CLASSIFICATION, CROSS_MODAL_GENERATION, MULTIMODAL_RETRIEVAL
    }

    public static class MultiModalInput {
        private final Map<ModalityType, Object> data;
        private final TaskType task;

        public MultiModalInput(Map<ModalityType, Object> data, TaskType task) {
            this.data = new HashMap<>(data);
            this.task = task;
        }

        public Map<ModalityType, Object> getData() { return Collections.unmodifiableMap(data); }
        public TaskType getTask() { return task; }
    }

    public static class MultiModalResult {
        private final FusedFeatures fusedFeatures;
        private final Object output;
        private final long timestamp;

        public MultiModalResult(FusedFeatures fusedFeatures, Object output, long timestamp) {
            this.fusedFeatures = fusedFeatures;
            this.output = output;
            this.timestamp = timestamp;
        }

        public FusedFeatures getFusedFeatures() { return fusedFeatures; }
        public Object getOutput() { return output; }
        public long getTimestamp() { return timestamp; }
    }

    public static class ModalityFeatures {
        private final ModalityType modality;
        private final float[] features;
        private final Map<String, Object> metadata;

        public ModalityFeatures(ModalityType modality, float[] features, Map<String, Object> metadata) {
            this.modality = modality;
            this.features = features.clone();
            this.metadata = new HashMap<>(metadata);
        }

        public ModalityType getModality() { return modality; }
        public float[] getFeatures() { return features.clone(); }
        public Map<String, Object> getMetadata() { return Collections.unmodifiableMap(metadata); }
    }

    public static class FusedFeatures {
        private final float[] fusedVector;
        private final Map<ModalityType, float[]> alignedFeatures;
        private final Map<ModalityType, float[]> attentionWeights;

        public FusedFeatures(float[] fusedVector, Map<ModalityType, float[]> alignedFeatures,
                            Map<ModalityType, float[]> attentionWeights) {
            this.fusedVector = fusedVector.clone();
            this.alignedFeatures = new HashMap<>(alignedFeatures);
            this.attentionWeights = new HashMap<>(attentionWeights);
        }

        public float[] getFusedVector() { return fusedVector.clone(); }
        public Map<ModalityType, float[]> getAlignedFeatures() { return Collections.unmodifiableMap(alignedFeatures); }
        public Map<ModalityType, float[]> getAttentionWeights() { return Collections.unmodifiableMap(attentionWeights); }
    }

    // 模态处理器接口
    public interface ModalityProcessor {
        ModalityFeatures extractFeatures(Object data);
    }

    // 文本处理器
    public static class TextProcessor implements ModalityProcessor {
        @Override
        public ModalityFeatures extractFeatures(Object data) {
            String text = data.toString();
            // 模拟文本特征提取
            float[] features = new float[768]; // BERT embedding size
            Random random = new Random(text.hashCode());
            for (int i = 0; i < features.length; i++) {
                features[i] = (float) (random.nextGaussian() * 0.1);
            }

            Map<String, Object> metadata = new HashMap<>();
            metadata.put("length", text.length());
            metadata.put("language", "zh-CN");

            return new ModalityFeatures(ModalityType.TEXT, features, metadata);
        }
    }

    // 图像处理器
    public static class ImageProcessor implements ModalityProcessor {
        @Override
        public ModalityFeatures extractFeatures(Object data) {
            // 模拟图像特征提取
            float[] features = new float[2048]; // ResNet feature size
            Random random = new Random(data.hashCode());
            for (int i = 0; i < features.length; i++) {
                features[i] = (float) (random.nextGaussian() * 0.1);
            }

            Map<String, Object> metadata = new HashMap<>();
            metadata.put("format", "JPEG");
            metadata.put("size", "224x224");

            return new ModalityFeatures(ModalityType.IMAGE, features, metadata);
        }
    }

    // 音频处理器
    public static class AudioProcessor implements ModalityProcessor {
        @Override
        public ModalityFeatures extractFeatures(Object data) {
            // 模拟音频特征提取
            float[] features = new float[1024]; // Audio feature size
            Random random = new Random(data.hashCode());
            for (int i = 0; i < features.length; i++) {
                features[i] = (float) (random.nextGaussian() * 0.1);
            }

            Map<String, Object> metadata = new HashMap<>();
            metadata.put("duration", "10.5s");
            metadata.put("sample_rate", "16000");

            return new ModalityFeatures(ModalityType.AUDIO, features, metadata);
        }
    }

    // 视频处理器
    public static class VideoProcessor implements ModalityProcessor {
        @Override
        public ModalityFeatures extractFeatures(Object data) {
            // 模拟视频特征提取
            float[] features = new float[3072]; // Video feature size
            Random random = new Random(data.hashCode());
            for (int i = 0; i < features.length; i++) {
                features[i] = (float) (random.nextGaussian() * 0.1);
            }

            Map<String, Object> metadata = new HashMap<>();
            metadata.put("fps", 30);
            metadata.put("duration", "60s");

            return new ModalityFeatures(ModalityType.VIDEO, features, metadata);
        }
    }

    // 测试示例
    public static void main(String[] args) throws Exception {
        MultiModalProcessorManager manager = new MultiModalProcessorManager();

        // 创建多模态输入
        Map<ModalityType, Object> inputData = new HashMap<>();
        inputData.put(ModalityType.TEXT, "这是一段关于猫的描述");
        inputData.put(ModalityType.IMAGE, "cat_image.jpg");
        inputData.put(ModalityType.AUDIO, "cat_meow.wav");

        MultiModalInput input = new MultiModalInput(inputData, TaskType.MULTIMODAL_CLASSIFICATION);

        // 处理多模态数据
        CompletableFuture<MultiModalResult> future = manager.process(input);
        MultiModalResult result = future.get(60, TimeUnit.SECONDS);

        // 输出结果
        System.out.println("=== 多模态处理结果 ===");
        System.out.println("输出: " + result.getOutput());
        System.out.println("处理时间戳: " + result.getTimestamp());

        FusedFeatures fusedFeatures = result.getFusedFeatures();
        System.out.printf("融合特征维度: %d%n", fusedFeatures.getFusedVector().length);

        System.out.println("\n=== 跨模态注意力权重 ===");
        for (Map.Entry<ModalityType, float[]> entry : fusedFeatures.getAttentionWeights().entrySet()) {
            System.out.printf("%s: %s%n", entry.getKey(), Arrays.toString(entry.getValue()));
        }
    }
}
```
