## Java锁机制与AI数据同步

### ⭐ 基础题 (1-30)

**1. Java中有哪些锁类型？各自的特点和适用场景是什么？**

**面试场景**：Java开发工程师面试，考察锁机制基础知识

**口语化答案**：
Java中的锁主要分为以下几个类型：

**1. 悲观锁 vs 乐观锁**：
- 悲观锁：假设冲突会发生，先加锁再操作（synchronized、ReentrantLock）
- 乐观锁：假设冲突不常发生，不加锁，冲突时重试（CAS、AtomicInteger）

**2. 可重入锁 vs 非可重入锁**：
- 可重入锁：同一线程可多次获取同一把锁（ReentrantLock）
- 非可重入锁：同一线程只能获取一次

**3. 公平锁 vs 非公平锁**：
- 公平锁：按请求顺序获取锁，吞吐量较低
- 非公平锁：可插队，吞吐量较高

**4. 共享锁 vs 排他锁**：
- 排他锁：同时只允许一个线程访问（写锁）
- 共享锁：允许多个线程同时访问（读锁）

在AI系统中的典型应用：
- 模型参数更新：使用乐观锁避免训练冲突
- 数据预处理：使用可重入锁简化编程
- 读写分离：使用读写锁优化数据访问

/**

public class JavaLockMechanismsDemo {

    /**
     * AI模型参数存储 - 演示不同锁机制
     */
    public static class AIModelParameterStore {
        // 模型参数存储
        private final Map<String, Double> parameters = new ConcurrentHashMap<>();
        private final ReadWriteLock rwLock = new ReentrantReadWriteLock();
        private final Lock readLock = rwLock.readLock();
        private final Lock writeLock = rwLock.writeLock();

        // 版本号 - 用于乐观锁
        private final AtomicLong version = new AtomicLong(0);

        // 统计信息
        private final AtomicLong readCount = new AtomicLong(0);
        private final AtomicLong writeCount = new AtomicLong(0);

        /**
         * 悲观锁读取参数
         */
        public double getParameterPessimistic(String paramName) {
            readLock.lock();
            try {
                readCount.incrementAndGet();
                return parameters.getOrDefault(paramName, 0.0);
            } finally {
                readLock.unlock();
            }
        }

        /**
         * 悲观锁更新参数
         */
        public void updateParameterPessimistic(String paramName, double value) {
            writeLock.lock();
            try {
                writeCount.incrementAndGet();
                parameters.put(paramName, value);
                version.incrementAndGet();
            } finally {
                writeLock.unlock();
            }
        }

        /**
         * 乐观锁读取参数
         */
        public ParameterVersion getParameterOptimistic(String paramName) {
            readLock.lock();
            try {
                double value = parameters.getOrDefault(paramName, 0.0);
                long currentVersion = version.get();
                return new ParameterVersion(value, currentVersion);
            } finally {
                readLock.unlock();
            }
        }

        /**
         * 乐观锁更新参数（CAS操作）
         */
        public boolean updateParameterOptimistic(String paramName, double newValue, long expectedVersion) {
            writeLock.lock();
            try {
                if (version.get() == expectedVersion) {
                    parameters.put(paramName, newValue);
                    version.incrementAndGet();
                    return true;
                }
                return false;
            } finally {
                writeLock.unlock();
            }
        }

        /**
         * 批量更新参数 - 演示可重入锁
         */
        public void batchUpdateParameters(Map<String, Double> newParams) {
            writeLock.lock();
            try {
                // 可重入：同一个线程可以多次获取写锁
                System.out.println("开始批量更新参数...");

                for (Map.Entry<String, Double> entry : newParams.entrySet()) {
                    updateSingleParameter(entry.getKey(), entry.getValue());
                }

                System.out.println("批量更新完成");
            } finally {
                writeLock.unlock();
            }
        }

        private void updateSingleParameter(String name, double value) {
            // 这里仍然持有写锁，演示可重入特性
            parameters.put(name, value);
        }

        /**
         * 获取所有参数的只读视图
         */
        public Map<String, Double> getAllParameters() {
            readLock.lock();
            try {
                return Collections.unmodifiableMap(new HashMap<>(parameters));
            } finally {
                readLock.unlock();
            }
        }

        /**
         * 获取统计信息
         */
        public LockStatistics getStatistics() {
            return new LockStatistics(readCount.get(), writeCount.get(), version.get());
        }

        /**
         * 参数版本类
         */
        public static class ParameterVersion {
            private final double value;
            private final long version;

            public ParameterVersion(double value, long version) {
                this.value = value;
                this.version = version;
            }

            public double getValue() { return value; }
            public long getVersion() { return version; }
        }

        /**
         * 锁统计信息类

        public static class LockStatistics {
            private final long readCount;
            private final long writeCount;
            private final long currentVersion;

            public LockStatistics(long readCount, long writeCount, long currentVersion) {
                this.readCount = readCount;
                this.writeCount = writeCount;
                this.currentVersion = currentVersion;
            }

            @Override
            public String toString() {
                return String.format("LockStats{reads=%d, writes=%d, version=%d}",
                    readCount, writeCount, currentVersion);
            }
        }

    /**
     * AI训练任务调度器 - 演示公平锁vs非公平锁

    public static class AITrainingScheduler {
        private final Lock fairLock = new ReentrantLock(true);   // 公平锁
        private final Lock unfairLock = new ReentrantLock(false); // 非公平锁

        private final Queue<TrainingTask> taskQueue = new LinkedList<>();
        private volatile boolean useFairLock = true;

        /**
         * 提交训练任务
         */
        public void submitTask(TrainingTask task) {
            Lock lock = useFairLock ? fairLock : unfairLock;

            lock.lock();
            try {
                taskQueue.offer(task);
                System.out.printf("%s 提交任务: %s (锁类型: %s)%n",
                    Thread.currentThread().getName(), task.getTaskId(),
                    useFairLock ? "公平锁" : "非公平锁");
            } finally {
                lock.unlock();
            }
        }

        /**
         * 获取下一个任务
         */
        public TrainingTask getNextTask() {
            Lock lock = useFairLock ? fairLock : unfairLock;

            lock.lock();
            try {
                TrainingTask task = taskQueue.poll();
                if (task != null) {
                    System.out.printf("%s 获取任务: %s (锁类型: %s)%n",
                        Thread.currentThread().getName(), task.getTaskId(),
                        useFairLock ? "公平锁" : "非公平锁");
                }
                return task;
            } finally {
                lock.unlock();
            }
        }

        /**
         * 切换锁类型
         */
        public void switchLockType(boolean useFair) {
            useFairLock = useFair;
            System.out.printf("切换到%s锁%n", useFair ? "公平" : "非公平");
        }

        /**
         * 训练任务类
         */
        public static class TrainingTask {
            private final String taskId;
            private final Runnable task;

            public TrainingTask(String taskId, Runnable task) {
                this.taskId = taskId;
                this.task = task;
            }

            public String getTaskId() { return taskId; }
            public void execute() { task.run(); }
        }
    }

    /**
     * AI模型推理缓存 - 演示StampedLock
     */
    public static class AIInferenceCache {
        private final StampedLock stampedLock = new StampedLock();
        private final Map<String, InferenceResult> cache = new ConcurrentHashMap<>();

        /**
         * 乐观读获取缓存
         */
        public InferenceResult getCachedResultOptimistic(String modelId, double[] input) {
            String cacheKey = generateCacheKey(modelId, input);
            long stamp = stampedLock.tryOptimisticRead();

            // 乐观读取
            InferenceResult result = cache.get(cacheKey);

            // 验证乐观读
            if (!stampedLock.validate(stamp)) {
                // 乐观读失败，降级为悲观读
                stamp = stampedLock.readLock();
                try {
                    result = cache.get(cacheKey);
                } finally {
                    stampedLock.unlockRead(stamp);
                }
            }

            return result;
        }

        /**
         * 悲观读获取缓存
         */
        public InferenceResult getCachedResultPessimistic(String modelId, double[] input) {
            String cacheKey = generateCacheKey(modelId, input);
            long stamp = stampedLock.readLock();

            try {
                return cache.get(cacheKey);
            } finally {
                stampedLock.unlockRead(stamp);
            }
        }

        /**
         * 更新缓存
         */
        public void putCachedResult(String modelId, double[] input, InferenceResult result) {
            String cacheKey = generateCacheKey(modelId, input);
            long stamp = stampedLock.writeLock();

            try {
                cache.put(cacheKey, result);
            } finally {
                stampedLock.unlockWrite(stamp);
            }
        }

        /**
         * 条件更新缓存
         */
        public boolean putIfAbsent(String modelId, double[] input, InferenceResult result) {
            String cacheKey = generateCacheKey(modelId, input);

            // 先尝试乐观读
            long stamp = stampedLock.tryOptimisticRead();
            InferenceResult existing = cache.get(cacheKey);

            if (stampedLock.validate(stamp) && existing != null) {
                return false; // 已存在
            }

            // 升级为写锁
            stamp = stampedLock.readLock();
            try {
                // 再次检查
                existing = cache.get(cacheKey);
                if (existing == null) {
                    long writeStamp = stampedLock.tryConvertToWriteLock(stamp);
                    if (writeStamp != 0L) {
                        try {
                            cache.put(cacheKey, result);
                            return true;
                        } finally {
                            stampedLock.unlockWrite(writeStamp);
                        }
                    } else {
                        // 转换失败，重新获取写锁
                        stampedLock.unlockRead(stamp);
                        stamp = stampedLock.writeLock();
                        try {
                            cache.put(cacheKey, result);
                            return true;
                        } finally {
                            stampedLock.unlockWrite(stamp);
                        }
                    }
                }
                return false;
            } finally {
                if (StampedLock.isReadLockStamp(stamp)) {
                    stampedLock.unlockRead(stamp);
                }
            }
        }

        private String generateCacheKey(String modelId, double[] input) {
            return modelId + "_" + Arrays.hashCode(input);
        }

        /**
         * 缓存统计
         */
        public CacheStats getStats() {
            return new CacheStats(cache.size());
        }

        /**
         * 推理结果类
         */
        public static class InferenceResult {
            private final double[] probabilities;
            private final long timestamp;

            public InferenceResult(double[] probabilities) {
                this.probabilities = probabilities;
                this.timestamp = System.currentTimeMillis();
            }

            public double[] getProbabilities() { return probabilities; }
            public long getTimestamp() { return timestamp; }
        }

        /**
         * 缓存统计类

        public static class CacheStats {
            private final int size;

            public CacheStats(int size) {
                this.size = size;

            @Override
            public String toString() {

    public static class LockPerformanceBenchmark {
        private static final int THREAD_COUNT = 8;
        private static final int OPERATIONS_PER_THREAD = 10000;

        /**
         * 对比不同锁的性能
         */
        public static void compareLockPerformance() {
            System.out.println("=== 锁性能对比测试 ===\n");

            // 测试synchronized
            long syncTime = testSynchronized();

            // 测试ReentrantLock
            long reentrantLockTime = testReentrantLock();

            // 测试ReadWriteLock
            long readWriteLockTime = testReadWriteLock();

            // 测试StampedLock
            long stampedLockTime = testStampedLock();

            // 测试AtomicInteger（无锁）
            long atomicTime = testAtomic();

            System.out.println("性能对比结果:");
            System.out.printf("synchronized:    %d ms%n", syncTime);
            System.out.printf("ReentrantLock:   %d ms%n", reentrantLockTime);
            System.out.printf("ReadWriteLock:   %d ms%n", readWriteLockTime);
            System.out.printf("StampedLock:     %d ms%n", stampedLockTime);
            System.out.printf("AtomicInteger:   %d ms%n", atomicTime);

            // 计算相对性能
            double baseline = (double) syncTime;
            System.out.printf("\n相对性能 (synchronized = 1.0x):%n");
            System.out.printf("ReentrantLock:   %.2fx%n", baseline / reentrantLockTime);
            System.out.printf("ReadWriteLock:   %.2fx%n", baseline / readWriteLockTime);
            System.out.printf("StampedLock:     %.2fx%n", baseline / stampedLockTime);
            System.out.printf("AtomicInteger:   %.2fx%n", baseline / atomicTime);
        }

        private static long testSynchronized() {
            Counter counter = new SynchronizedCounter();
            return runBenchmark("synchronized", counter);
        }

        private static long testReentrantLock() {
            Counter counter = new ReentrantLockCounter();
            return runBenchmark("ReentrantLock", counter);
        }

        private static long testReadWriteLock() {
            Counter counter = new ReadWriteLockCounter();
            return runBenchmark("ReadWriteLock", counter);
        }

        private static long testStampedLock() {
            Counter counter = new StampedLockCounter();
            return runBenchmark("StampedLock", counter);
        }

        private static long testAtomic() {
            Counter counter = new AtomicCounter();
            return runBenchmark("Atomic", counter);
        }

        private static long runBenchmark(String lockType, Counter counter) {
            ExecutorService executor = Executors.newFixedThreadPool(THREAD_COUNT);
            CountDownLatch latch = new CountDownLatch(THREAD_COUNT);

            long startTime = System.currentTimeMillis();

            for (int i = 0; i < THREAD_COUNT; i++) {
                executor.submit(() -> {
                    for (int j = 0; j < OPERATIONS_PER_THREAD; j++) {
                        counter.increment();
                        if (j % 100 == 0) {
                            counter.get();
                        }
                    }
                    latch.countDown();
                });
            }

            try {
                latch.await();
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }

            long endTime = System.currentTimeMillis();
            executor.shutdown();

            System.out.printf("%s 最终计数值: %d%n", lockType, counter.get());
            return endTime - startTime;
        }

        /**
         * 计数器接口
         */
        private interface Counter {
            void increment();
            long get();
        }

        private static class SynchronizedCounter implements Counter {
            private long count = 0;

            @Override
            public synchronized void increment() {
                count++;
            }

            @Override
            public synchronized long get() {
                return count;
            }
        }

        private static class ReentrantLockCounter implements Counter {
            private final ReentrantLock lock = new ReentrantLock();
            private long count = 0;

            @Override
            public void increment() {
                lock.lock();
                try {
                    count++;
                } finally {
                    lock.unlock();
                }
            }

            @Override
            public long get() {
                lock.lock();
                try {
                    return count;
                } finally {
                    lock.unlock();
                }
            }
        }

        private static class ReadWriteLockCounter implements Counter {
            private final ReadWriteLock rwLock = new ReentrantReadWriteLock();
            private volatile long count = 0;

            @Override
            public void increment() {
                rwLock.writeLock().lock();
                try {
                    count++;
                } finally {
                    rwLock.writeLock().unlock();
                }
            }

            @Override
            public long get() {
                rwLock.readLock().lock();
                try {
                    return count;
                } finally {
                    rwLock.readLock().unlock();
                }
            }
        }

        private static class StampedLockCounter implements Counter {
            private final StampedLock stampedLock = new StampedLock();
            private volatile long count = 0;

            @Override
            public void increment() {
                long stamp = stampedLock.writeLock();
                try {
                    count++;
                } finally {
                    stampedLock.unlockWrite(stamp);
                }
            }

            @Override
            public long get() {
                long stamp = stampedLock.tryOptimisticRead();
                long result = count;

                if (!stampedLock.validate(stamp)) {
                    stamp = stampedLock.readLock();
                    try {
                        result = count;
                    } finally {
                        stampedLock.unlockRead(stamp);
                    }
                }

                return result;
            }
        }

        private static class AtomicCounter implements Counter {
            private final AtomicInteger count = new AtomicInteger(0);

            @Override
            public void increment() {
                count.incrementAndGet();
            }

            @Override
            public long get() {
                return count.get();
            }
        }
    }

    private static void demonstrateTrainingScheduler() {
        System.out.println("\n=== AI训练任务调度演示 ===\n");

        AITrainingScheduler scheduler = new AITrainingScheduler();

        // 创建训练任务
        List<AITrainingScheduler.TrainingTask> tasks = new ArrayList<>();
        for (int i = 0; i < 10; i++) {
            final int taskId = i;
            Runnable task = () -> {
                try {
                    Thread.sleep(100 + (int)(Math.random() * 200));
                    System.out.printf("任务 %d 执行完成%n", taskId);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            };
            tasks.add(new AITrainingScheduler.TrainingTask("task_" + taskId, task));
        }

        // 提交任务
        ExecutorService executor = Executors.newFixedThreadPool(3);

        // 提交者线程
        executor.submit(() -> {
            for (AITrainingScheduler.TrainingTask task : tasks) {
                scheduler.submitTask(task);
                try {
                    Thread.sleep(50);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            }
        });

        // 消费者线程
        for (int i = 0; i < 2; i++) {
            executor.submit(() -> {
                while (true) {
                    AITrainingScheduler.TrainingTask task = scheduler.getNextTask();
                    if (task != null) {
                        task.execute();
                    } else {
                        try {
                            Thread.sleep(100);
                        } catch (InterruptedException e) {
                            Thread.currentThread().interrupt();
                            break;
                        }
                    }
                }
            });
        }

        try {
            Thread.sleep(3000);
            scheduler.switchLockType(false); // 切换到非公平锁
            Thread.sleep(2000);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        executor.shutdown();
    }

## AQS框架与自定义同步器

### ⭐⭐ 进阶题 (31-70)

**31. AQS(AbstractQueuedSynchronizer)的原理是什么？如何基于AQS实现自定义同步器？**

**面试场景**：高级Java开发面试，考察AQS底层原理和自定义同步器实现

**口语化答案**：
AQS是Java并发包的核心基础框架，它提供了一个基于FIFO等待队列的同步器框架。

**AQS核心原理**：
1. **状态管理**：使用volatile int state维护同步状态
2. **等待队列**：基于CLH队列的FIFO等待线程队列
3. **独占/共享模式**：支持独占（如ReentrantLock）和共享（如Semaphore）模式
4. **条件变量**：支持Condition条件等待

**核心方法**：
- `tryAcquire(int arg)`：尝试获取锁（独占模式）
- `tryRelease(int arg)`：尝试释放锁（独占模式）
- `tryAcquireShared(int arg)`：尝试获取锁（共享模式）
- `tryReleaseShared(int arg)`：尝试释放锁（共享模式）
- `isHeldExclusively()`：是否独占持有

在AI系统中可以基于AQS实现：
- 资源池管理器
- 模型训练队列控制
- 数据流门控同步器
- 自定义限流器

         * AQS同步器实现
         */
        private static class Sync extends AbstractQueuedSynchronizer {
            public Sync(int permits) {
                setState(permits);
            }

            @Override
            protected int tryAcquireShared(int acquires) {
                for (;;) {
                    int available = getState();
                    int remaining = available - acquires;
                    if (remaining < 0 || compareAndSetState(available, remaining)) {
                        return remaining;
                    }
                }
            }

            @Override
            protected boolean tryReleaseShared(int releases) {
                for (;;) {
                    int current = getState();
                    int newCount = current + releases;
                    if (newCount < current) { // overflow
                        throw new Error("Maximum permit count exceeded");
                    }
                    if (compareAndSetState(current, newCount)) {
                        return true;
                    }
                }
            }

            public int getAvailablePermits() {
                return getState();
            }
        }

        }

         * 带超时的等待
         */
        public int await(long timeout, TimeUnit unit) throws InterruptedException,
                BrokenBarrierException, TimeoutException {
            return sync.dowait(true, unit.toNanos(timeout));
        }

        /**
         * 重置屏障
         */
        public void reset() {
            sync.reset();
        }

         * 检查屏障是否已损坏
         */
        public boolean isBroken() {
            return sync.isBroken();
        }

         */

            return sync.getWaiters();

         * 获取参与方总数

            return parties;

        private static class Sync extends AbstractQueuedSynchronizer {
            private static final int WAITING = -1;
            private static final int TRIPPED = 1;
            private static final int BROKEN = 2;

            private final int parties;
            private volatile int waiters;
            private volatile boolean broken;

            public Sync(int parties) {
                this.parties = parties;
                this.waiters = 0;
                this.broken = false;
            }

            int dowait(boolean timed, long nanos) throws InterruptedException,
                    BrokenBarrierException, TimeoutException {
                final ReentrantLock lock = new ReentrantLock();
                lock.lock();
                try {
                    if (broken) {
                        throw new BrokenBarrierException();
                    }

                    int index = --waiters;
                    if (index == 0) { // 最后一个到达
                        setState(TRIPPED);
                        return 0;
                    }

                    // 等待
                    for (;;) {
                        if (Thread.interrupted()) {
                            setBroken();
                            throw new InterruptedException();
                        }

                        int state = getState();
                        if (state == TRIPPED) {
                            return index;
                        }
                        if (state == BROKEN) {
                            throw new BrokenBarrierException();
                        }

                        if (timed && nanos <= 0L) {
                            setBroken();
                            throw new TimeoutException();
                        }

                        // 简化实现，实际应该使用Condition
                        if (!timed) {
                            lock.wait();
                        } else {
                            long nanosRemaining = nanos;
                            long startTime = System.nanoTime();
                            lock.wait(nanosRemaining / 1_000_000, (int) (nanosRemaining % 1_000_000));
                            nanosRemaining -= (System.nanoTime() - startTime);
                            nanos = nanosRemaining;
                        }
                    }
                } finally {
                    lock.unlock();
                }
            }

            private void setBroken() {
                broken = true;
                setState(BROKEN);
            }

            boolean isBroken() {
                return broken;
            }

            int getWaiters() {
                return waiters;
            }

            void reset() {
                broken = false;
                waiters = parties;
                setState(0);
            }
        }
    }

     * AI资源池管理器 - 基于AQS实现可重入资源池

    public static class AIResourcePool<T> {
        private final Sync sync;
        private final Set<T> availableResources;
        private final Map<T, Integer> resourceUsage; // 记录资源使用次数

        public AIResourcePool(Set<T> resources) {
            this.availableResources = new ConcurrentHashMap<>(resources);
            this.resourceUsage = new ConcurrentHashMap<>();
            this.sync = new Sync(resources.size());

            // 初始化资源使用次数
            for (T resource : resources) {
                resourceUsage.put(resource, 0);
            }
        }

        /**
         * 获取资源
         */
        public T acquire() throws InterruptedException {
            sync.acquire(1);
            return takeResource();
        }

        /**
         * 尝试获取资源（非阻塞）
         */
        public T tryAcquire() {
            if (sync.tryAcquire(1)) {
                return takeResource();
            }
            return null;
        }

        /**
         * 释放资源
         */
        public void release(T resource) {
            if (resource != null) {
                putResource(resource);
                sync.release(1);
            }
        }

        /**
         * 获取可用资源数量
         */
        public int availableCount() {
            return availableResources.size();
        }

        /**
         * 获取总资源数量
         */
        public int totalCount() {
            return resourceUsage.size();
        }

        /**
         * 获取资源使用统计
         */
        public Map<T, Integer> getResourceUsageStats() {
            return new HashMap<>(resourceUsage);
        }

        private T takeResource() {
            Iterator<T> iterator = availableResources.iterator();
            if (iterator.hasNext()) {
                T resource = iterator.next();
                iterator.remove();

                // 更新使用次数
                resourceUsage.merge(resource, 1, Integer::sum);

                return resource;
            }
            throw new IllegalStateException("No available resource");
        }

        private void putResource(T resource) {
            availableResources.add(resource);
        }

        private static class Sync extends AbstractQueuedSynchronizer {
            public Sync(int permits) {
                setState(permits);
            }

            @Override
            protected boolean tryAcquire(int acquires) {
                if (getState() >= acquires) {
                    if (compareAndSetState(getState(), getState() - acquires)) {
                        return true;
                    }
                }
                return false;
            }

            @Override
            protected boolean tryRelease(int releases) {
                return compareAndSetState(getState(), getState() + releases);
            }

            @Override
            protected boolean isHeldExclusively() {
                return true;
            }

            Condition newCondition() {
                return new ConditionObject();
            }
        }

设计高并发AI模型参数服务器需要考虑以下几个核心要素：

**1. 数据结构选择**：

- **无锁数据结构**：使用ConcurrentHashMap等并发集合

**3. 网络通信优化**：
- **批量操作**：支持批量参数读写

**4. 容错和恢复**：
- **心跳检测**：监控节点健康状态

/**
 * 高并发AI模型参数服务器实现

public class AIParameterServer {

    /**
     * 参数分片管理器
     */
    public static class ParameterShardManager {
        private final int shardCount;
        private final List<ParameterShard> shards;
        private final ConsistentHashRing<String> hashRing;

        public ParameterShardManager(int shardCount) {
            this.shardCount = shardCount;
            this.shards = new ArrayList<>();
            this.hashRing = new ConsistentHashRing<>();

            // 初始化分片
            for (int i = 0; i < shardCount; i++) {
                ParameterShard shard = new ParameterShard("shard_" + i);
                shards.add(shard);
                hashRing.addNode("shard_" + i);
            }
        }

        /**
         * 存储参数
         */
        public CompletableFuture<Void> storeParameter(String modelId, String paramName,
                                                   float[] values, long version) {
            String shardId = getShardId(modelId, paramName);
            ParameterShard shard = getShard(shardId);

            ParameterData data = new ParameterData(modelId, paramName, values, version);
            return shard.storeParameter(data);
        }

        /**
         * 获取参数
         */
        public CompletableFuture<ParameterData> getParameter(String modelId, String paramName) {
            String shardId = getShardId(modelId, paramName);
            ParameterShard shard = getShard(shardId);

            return shard.getParameter(modelId, paramName);
        }

        /**
         * 批量获取参数
         */
        public CompletableFuture<List<ParameterData>> getParameters(String modelId,
                                                                 List<String> paramNames) {
            // 按分片分组参数请求
            Map<String, List<String>> shardParams = new HashMap<>();

            for (String paramName : paramNames) {
                String shardId = getShardId(modelId, paramName);
                shardParams.computeIfAbsent(shardId, k -> new ArrayList<>()).add(paramName);
            }

            // 并行从各分片获取参数
            List<CompletableFuture<List<ParameterData>>> futures = shardParams.entrySet().stream()
                .map(entry -> {
                    String shardId = entry.getKey();
                    List<String> params = entry.getValue();
                    ParameterShard shard = getShard(shardId);
                    return shard.getParameters(modelId, params);
                })
                .collect(Collectors.toList());

            // 合并结果
            return CompletableFuture.allOf(futures.toArray(new CompletableFuture[0]))
                .thenApply(v -> {
                    List<ParameterData> result = new ArrayList<>();
                    for (CompletableFuture<List<ParameterData>> future : futures) {
                        try {
                            result.addAll(future.get());
                        } catch (Exception e) {
                            throw new RuntimeException("获取参数失败", e);
                        }
                    }
                    return result;
                });
        }

        /**
         * 获取分片ID
         */
        private String getShardId(String modelId, String paramName) {
            String key = modelId + ":" + paramName;
            return hashRing.getNode(key);
        }

        /**
         * 获取分片
         */
        private ParameterShard getShard(String shardId) {
            return shards.stream()
                .filter(shard -> shard.getShardId().equals(shardId))
                .findFirst()
                .orElseThrow(() -> new IllegalArgumentException("分片不存在: " + shardId));
        }

        /**
         * 获取分片统计信息
         */
        public ShardStatistics getShardStatistics() {
            Map<String, ParameterShard.ShardStats> stats = new HashMap<>();
            for (ParameterShard shard : shards) {
                stats.put(shard.getShardId(), shard.getStatistics());
            }
            return new ShardStatistics(stats);
        }
    }

    /**
     * 参数分片
     */
    public static class ParameterShard {
        private final String shardId;
        private final ConcurrentHashMap<String, ParameterData> parameters;
        private final ReadWriteLock versionLock;
        private final AtomicLong totalOperations;
        private final AtomicInteger cacheHits;
        private final AtomicInteger cacheMisses;

        // 参数缓存
        private final Cache<String, ParameterData> paramCache;

        public ParameterShard(String shardId) {
            this.shardId = shardId;
            this.parameters = new ConcurrentHashMap<>();
            this.versionLock = new ReentrantReadWriteLock();
            this.totalOperations = new AtomicLong(0);
            this.cacheHits = new AtomicInteger(0);
            this.cacheMisses = new AtomicInteger(0);

            // 创建LRU缓存
            this.paramCache = CacheBuilder.newBuilder()
                .maximumSize(10000)
                .expireAfterAccess(30, TimeUnit.MINUTES)
                .recordStats()
                .build();
        }

        /**
         * 存储参数
         */
        public CompletableFuture<Void> storeParameter(ParameterData data) {
            return CompletableFuture.runAsync(() -> {
                totalOperations.incrementAndGet();

                // 检查版本冲突
                versionLock.readLock().lock();
                try {
                    ParameterData existing = parameters.get(data.getKey());
                    if (existing != null && existing.getVersion() > data.getVersion()) {
                        throw new IllegalArgumentException("版本冲突，现有版本更新: " +
                            existing.getVersion() + " vs " + data.getVersion());
                    }
                } finally {
                    versionLock.readLock().unlock();
                }

                // 更新参数
                versionLock.writeLock().lock();
                try {
                    ParameterData oldData = parameters.put(data.getKey(), data);
                    paramCache.put(data.getKey(), data);

                    // 如果是更新操作，可能需要通知观察者
                    if (oldData != null && !oldData.getVersion().equals(data.getVersion())) {
                        notifyParameterUpdate(oldData, data);
                    }
                } finally {
                    versionLock.writeLock().unlock();
                }
            });
        }

        /**
         * 获取参数
         */
        public CompletableFuture<ParameterData> getParameter(String modelId, String paramName) {
            return CompletableFuture.supplyAsync(() -> {
                totalOperations.incrementAndGet();
                String key = modelId + ":" + paramName;

                // 先从缓存获取
                try {
                    ParameterData cached = paramCache.get(key, () -> {
                        cacheMisses.incrementAndGet();
                        return loadFromStorage(key);
                    });

                    if (cached != null) {
                        cacheHits.incrementAndGet();
                        return cached.clone();
                    }
                } catch (Exception e) {
                    // 缓存加载失败，直接从存储加载
                    cacheMisses.incrementAndGet();
                }

                return loadFromStorage(key);
            });
        }

        /**
         * 批量获取参数
         */
        public CompletableFuture<List<ParameterData>> getParameters(String modelId,
                                                                   List<String> paramNames) {
            return CompletableFuture.supplyAsync(() -> {
                List<ParameterData> results = new ArrayList<>();

                for (String paramName : paramNames) {
                    try {
                        ParameterData data = getParameter(modelId, paramName).get();
                        if (data != null) {
                            results.add(data);
                        }
                    } catch (Exception e) {
                        System.err.printf("获取参数失败 %s: %s%n", paramName, e.getMessage());
                    }
                }

                return results;
            });
        }

        /**
         * 从存储加载参数
         */
        private ParameterData loadFromStorage(String key) {
            ParameterData data = parameters.get(key);
            if (data != null) {
                return data.clone();
            }
            return null;
        }

    public static class ConsistentHashRing<T> {
        private final TreeMap<Integer, T> ring = new TreeMap<>();
        private final int virtualNodes = 150; // 虚拟节点数量

        public void addNode(T node) {
            for (int i = 0; i < virtualNodes; i++) {
                String virtualNodeName = node.toString() + ":" + i;
                int hash = virtualNodeName.hashCode();
                ring.put(hash, node);
            }
        }

        public T getNode(String key) {
            if (ring.isEmpty()) {
                throw new IllegalStateException("哈希环为空");
            }

            int hash = key.hashCode();
            Map.Entry<Integer, T> entry = ring.ceilingEntry(hash);

            if (entry == null) {
                entry = ring.firstEntry(); // 环绕
            }

            return entry.getValue();
        }

        public int getNodeCount() {
            return ring.size() / virtualNodes;
        }
    }
