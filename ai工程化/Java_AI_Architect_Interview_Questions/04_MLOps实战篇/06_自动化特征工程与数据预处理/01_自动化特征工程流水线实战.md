### 题目1：企业级自动化特征工程平台 ⭐⭐⭐⭐⭐
**问题**：设计并实现一个企业级自动化特征工程平台，支持自动特征生成、特征选择、特征存储和特征监控等完整功能。

**核心架构组件**：
1. **自动特征生成器** - 统计特征、时序特征、交叉特征生成
2. **特征选择引擎** - 单变量选择、模型选择、递归特征消除
3. **特征存储系统** - 特征注册表、版本管理、元数据管理
4. **特征监控系统** - 特征漂移检测、性能监控、质量评估

import java.util.*;
import java.util.concurrent.*;
import java.util.stream.*;
import java.time.*;

/**
 * 自动化特征工程平台
 */
public class AutoFeatureEngineeringPlatform {

    // 特征工程平台主控制器
    public static class FeatureEngineeringPlatform {
        private final AutoFeatureGenerator featureGenerator;
        private final FeatureSelector featureSelector;
        private final FeatureRegistry featureRegistry;
        private final FeatureMonitor featureMonitor;
        private final FeatureStore featureStore;
        private final ExecutorService executorService;

        public FeatureEngineeringPlatform() {
            this.featureGenerator = new AutoFeatureGenerator();
            this.featureSelector = new FeatureSelector();
            this.featureRegistry = new FeatureRegistry();
            this.featureMonitor = new FeatureMonitor();
            this.featureStore = new FeatureStore();
            this.executorService = Executors.newFixedThreadPool(20);

            initializePlatform();
        }

        private void initializePlatform() {
            // 加载特征配置
            featureGenerator.loadFeatureConfigs();

            // 启动监控任务
            featureMonitor.startMonitoring();

            System.out.println("自动化特征工程平台启动完成");
        }

        // 执行完整的特征工程流水线
        public CompletableFuture<FeatureEngineeringResult> processFeatures(Dataset dataset,
                                                                         FeatureEngineeringConfig config) {
            return CompletableFuture.supplyAsync(() -> {
                try {
                    long startTime = System.currentTimeMillis();

                    // 1. 数据预处理
                    Dataset preprocessedData = preprocessData(dataset, config);

                    // 2. 自动特征生成
                    GeneratedFeatures generatedFeatures = featureGenerator.generateFeatures(preprocessedData, config);

                    // 3. 特征选择
                    SelectedFeatures selectedFeatures = featureSelector.selectFeatures(
                        preprocessedData, generatedFeatures, config);

                    // 4. 特征验证
                    FeatureValidationResult validation = validateFeatures(selectedFeatures, preprocessedData);

                    // 5. 特征注册
                    List<RegisteredFeature> registeredFeatures = registerFeatures(selectedFeatures, validation);

                    // 6. 特征存储
                    storeFeatures(registeredFeatures, preprocessedData);

                    // 7. 创建结果
                    FeatureEngineeringResult result = new FeatureEngineeringResult(
                        dataset.getId(),
                        registeredFeatures,
                        validation,
                        System.currentTimeMillis() - startTime,
                        Instant.now()
                    );

                    // 8. 启动特征监控
                    featureMonitor.monitorFeatures(registeredFeatures);

                    return result;

                } catch (Exception e) {
                    System.err.printf("特征工程处理失败: %s%n", e.getMessage());
                    return createErrorResult(dataset, e);
                }
            }, executorService);
        }

        private Dataset preprocessData(Dataset dataset, FeatureEngineeringConfig config) {
            DataPreprocessor preprocessor = new DataPreprocessor(config.getPreprocessingConfig());
            return preprocessor.process(dataset);
        }

        private FeatureValidationResult validateFeatures(SelectedFeatures features, Dataset dataset) {
            FeatureValidator validator = new FeatureValidator();
            return validator.validate(features, dataset);
        }

        private List<RegisteredFeature> registerFeatures(SelectedFeatures features, FeatureValidationResult validation) {
            List<RegisteredFeature> registeredFeatures = new ArrayList<>();

            for (Feature feature : features.getFeatures()) {
                if (validation.isValidFeature(feature.getName())) {
                    RegisteredFeature registered = featureRegistry.register(feature);
                    registeredFeatures.add(registered);
                }
            }

            return registeredFeatures;
        }

        private void storeFeatures(List<RegisteredFeature> features, Dataset dataset) {
            for (RegisteredFeature feature : features) {
                featureStore.store(feature, dataset);
            }
        }

        private FeatureEngineeringResult createErrorResult(Dataset dataset, Exception e) {
            return new FeatureEngineeringResult(
                dataset.getId(),
                Collections.emptyList(),
                new FeatureValidationResult(false, Collections.emptyList(), Collections.emptyList()),
                0,
                Instant.now()
            );
        }

        // 获取特征
        public CompletableFuture<Map<String, List<Double>>> getFeatures(String featureSetId, Dataset dataset) {
            return CompletableFuture.supplyAsync(() -> {
                Map<String, List<Double>> features = new HashMap<>();

                // 从特征存储中获取特征
                List<RegisteredFeature> registeredFeatures = featureRegistry.getFeatureSet(featureSetId);

                for (RegisteredFeature feature : registeredFeatures) {
                    List<Double> values = featureStore.getFeatureValues(feature, dataset);
                    features.put(feature.getName(), values);
                }

                return features;
            }, executorService);
        }
    }

    // 自动特征生成器
    public static class AutoFeatureGenerator {
        private final Map<FeatureType, FeatureExtractor> extractors;
        private final List<FeatureConfig> featureConfigs;

        public AutoFeatureGenerator() {
            this.extractors = new HashMap<>();
            this.featureConfigs = new ArrayList<>();

            initializeExtractors();
        }

        private void initializeExtractors() {
            extractors.put(FeatureType.STATISTICAL, new StatisticalFeatureExtractor());
            extractors.put(FeatureType.TEMPORAL, new TemporalFeatureExtractor());
            extractors.put(FeatureType.INTERACTION, new InteractionFeatureExtractor());
            extractors.put(FeatureType.POLYNOMIAL, new PolynomialFeatureExtractor());
            extractors.put(FeatureType.TEXT, new TextFeatureExtractor());
            extractors.put(FeatureType.CATEGORY, new CategoryFeatureExtractor());
        }

        public void loadFeatureConfigs() {
            // 加载特征配置
            featureConfigs.add(new FeatureConfig("mean", FeatureType.STATISTICAL, Map.of("operation", "mean")));
            featureConfigs.add(new FeatureConfig("std", FeatureType.STATISTICAL, Map.of("operation", "std")));
            featureConfigs.add(new FeatureConfig("trend", FeatureType.TEMPORAL, Map.of("window", 7)));
            featureConfigs.add(new FeatureConfig("interaction_ab", FeatureType.INTERACTION, Map.of("features", Arrays.asList("a", "b"))));
        }

        public GeneratedFeatures generateFeatures(Dataset dataset, FeatureEngineeringConfig config) {
            List<Feature> features = new ArrayList<>();

            // 基础特征提取
            features.addAll(extractBasicFeatures(dataset));

            // 统计特征
            if (config.isEnableStatisticalFeatures()) {
                features.addAll(extractStatisticalFeatures(dataset));
            }

            // 时序特征
            if (config.isEnableTemporalFeatures()) {
                features.addAll(extractTemporalFeatures(dataset));
            }

            // 交互特征
            if (config.isEnableInteractionFeatures()) {
                features.addAll(extractInteractionFeatures(dataset));
            }

            // 多项式特征
            if (config.isEnablePolynomialFeatures()) {
                features.addAll(extractPolynomialFeatures(dataset));
            }

            // 文本特征
            if (config.isEnableTextFeatures()) {
                features.addAll(extractTextFeatures(dataset));
            }

            // 分类特征编码
            if (config.isEnableCategoryFeatures()) {
                features.addAll(extractCategoryFeatures(dataset));
            }

            return new GeneratedFeatures(features);
        }

        private List<Feature> extractBasicFeatures(Dataset dataset) {
            List<Feature> features = new ArrayList<>();

            for (String column : dataset.getColumnNames()) {
                List<Object> values = dataset.getColumn(column);
                if (values.isEmpty()) continue;

                // 基础统计特征
                features.addAll(extractBasicStatsForColumn(column, values));
            }

            return features;
        }

        private List<Feature> extractBasicStatsForColumn(String columnName, List<Object> values) {
            List<Feature> features = new ArrayList<>();

            // 数值型特征
            if (isNumericColumn(values)) {
                List<Double> numericValues = convertToNumeric(values);

                // 均值
                double mean = numericValues.stream().mapToDouble(Double::doubleValue).average().orElse(0);
                features.add(new Feature(columnName + "_mean", mean, FeatureType.STATISTICAL));

                // 标准差
                double variance = numericValues.stream()
                    .mapToDouble(v -> Math.pow(v - mean, 2))
                    .average().orElse(0);
                double std = Math.sqrt(variance);
                features.add(new Feature(columnName + "_std", std, FeatureType.STATISTICAL));

                // 最值
                double min = numericValues.stream().mapToDouble(Double::doubleValue).min().orElse(0);
                double max = numericValues.stream().mapToDouble(Double::doubleValue).max().orElse(0);
                features.add(new Feature(columnName + "_min", min, FeatureType.STATISTICAL));
                features.add(new Feature(columnName + "_max", max, FeatureType.STATISTICAL));

                // 中位数
                double median = calculateMedian(numericValues);
                features.add(new Feature(columnName + "_median", median, FeatureType.STATISTICAL));

                // 偏度
                double skewness = calculateSkewness(numericValues, mean, std);
                features.add(new Feature(columnName + "_skewness", skewness, FeatureType.STATISTICAL));

                // 峰度
                double kurtosis = calculateKurtosis(numericValues, mean, std);
                features.add(new Feature(columnName + "_kurtosis", kurtosis, FeatureType.STATISTICAL));
            }

            // 分类特征
            else {
                // 唯一值数量
                long uniqueCount = values.stream().distinct().count();
                features.add(new Feature(columnName + "_unique_count", uniqueCount, FeatureType.CATEGORY));

                // 缺失值数量
                long missingCount = values.stream().filter(Objects::isNull).count();
                features.add(new Feature(columnName + "_missing_count", missingCount, FeatureType.STATISTICAL));
            }

            return features;
        }

        private List<Feature> extractStatisticalFeatures(Dataset dataset) {
            StatisticalFeatureExtractor extractor = (StatisticalFeatureExtractor) extractors.get(FeatureType.STATISTICAL);
            return extractor.extract(dataset);
        }

        private List<Feature> extractTemporalFeatures(Dataset dataset) {
            TemporalFeatureExtractor extractor = (TemporalFeatureExtractor) extractors.get(FeatureType.TEMPORAL);
            return extractor.extract(dataset);
        }

        private List<Feature> extractInteractionFeatures(Dataset dataset) {
            InteractionFeatureExtractor extractor = (InteractionFeatureExtractor) extractors.get(FeatureType.INTERACTION);
            return extractor.extract(dataset);
        }

        private List<Feature> extractPolynomialFeatures(Dataset dataset) {
            PolynomialFeatureExtractor extractor = (PolynomialFeatureExtractor) extractors.get(FeatureType.POLYNOMIAL);
            return extractor.extract(dataset);
        }

        private List<Feature> extractTextFeatures(Dataset dataset) {
            TextFeatureExtractor extractor = (TextFeatureExtractor) extractors.get(FeatureType.TEXT);
            return extractor.extract(dataset);
        }

        private List<Feature> extractCategoryFeatures(Dataset dataset) {
            CategoryFeatureExtractor extractor = (CategoryFeatureExtractor) extractors.get(FeatureType.CATEGORY);
            return extractor.extract(dataset);
        }

        private boolean isNumericColumn(List<Object> values) {
            return values.stream()
                .filter(Objects::nonNull)
                .allMatch(v -> v instanceof Number);
        }

        private List<Double> convertToNumeric(List<Object> values) {
            return values.stream()
                .filter(Objects::nonNull)
                .map(v -> ((Number) v).doubleValue())
                .collect(Collectors.toList());
        }

        private double calculateMedian(List<Double> values) {
            List<Double> sorted = values.stream().sorted().collect(Collectors.toList());
            int size = sorted.size();
            if (size == 0) return 0;
            if (size % 2 == 0) {
                return (sorted.get(size/2 - 1) + sorted.get(size/2)) / 2.0;
            } else {
                return sorted.get(size/2);
            }
        }

        private double calculateSkewness(List<Double> values, double mean, double std) {
            if (std == 0) return 0;
            double n = values.size();
            double sum = values.stream()
                .mapToDouble(v -> Math.pow((v - mean) / std, 3))
                .sum();
            return (n / ((n - 1) * (n - 2))) * sum;
        }

        private double calculateKurtosis(List<Double> values, double mean, double std) {
            if (std == 0) return 0;
            double n = values.size();
            double sum = values.stream()
                .mapToDouble(v -> Math.pow((v - mean) / std, 4))
                .sum();
            return (n * (n + 1) / ((n - 1) * (n - 2) * (n - 3))) * sum -
                   (3 * (n - 1) * (n - 1) / ((n - 2) * (n - 3)));
        }
    }

    // 特征选择器
    public static class FeatureSelector {
        private final UnivariateSelector univariateSelector;
        private final ModelBasedSelector modelBasedSelector;
        private final RecursiveFeatureEliminator rfeSelector;

        public FeatureSelector() {
            this.univariateSelector = new UnivariateSelector();
            this.modelBasedSelector = new ModelBasedSelector();
            this.rfeSelector = new RecursiveFeatureEliminator();
        }

        public SelectedFeatures selectFeatures(Dataset dataset, GeneratedFeatures generatedFeatures,
                                             FeatureEngineeringConfig config) {
            List<Feature> allFeatures = new ArrayList<>();

            // 添加原始特征
            allFeatures.addAll(generatedFeatures.getOriginalFeatures());

            // 添加生成的特征
            allFeatures.addAll(generatedFeatures.getFeatures());

            List<Feature> selectedFeatures = new ArrayList<>();

            // 1. 移除低方差特征
            selectedFeatures = removeLowVarianceFeatures(allFeatures, dataset);

            // 2. 单变量特征选择
            if (config.isEnableUnivariateSelection()) {
                selectedFeatures = univariateSelector.select(selectedFeatures, dataset, config.getTopK());
            }

            // 3. 基于模型的特征选择
            if (config.isEnableModelBasedSelection()) {
                selectedFeatures = modelBasedSelector.select(selectedFeatures, dataset, config.getTopK());
            }

            // 4. 递归特征消除
            if (config.isEnableRFE()) {
                selectedFeatures = rfeSelector.select(selectedFeatures, dataset, config.getTopK());
            }

            return new SelectedFeatures(selectedFeatures);
        }

        private List<Feature> removeLowVarianceFeatures(List<Feature> features, Dataset dataset) {
            List<Feature> filteredFeatures = new ArrayList<>();

            for (Feature feature : features) {
                List<Double> values = dataset.getColumn(feature.getName()).stream()
                    .filter(Objects::nonNull)
                    .map(v -> ((Number) v).doubleValue())
                    .collect(Collectors.toList());

                if (values.isEmpty()) {
                    filteredFeatures.add(feature);
                    continue;
                }

                // 计算方差
                double mean = values.stream().mapToDouble(Double::doubleValue).average().orElse(0);
                double variance = values.stream()
                    .mapToDouble(v -> Math.pow(v - mean, 2))
                    .average().orElse(0);

                // 方差阈值
                double varianceThreshold = 1e-8;
                if (variance > varianceThreshold) {
                    filteredFeatures.add(feature);
                }
            }

            return filteredFeatures;
        }

        // 单变量选择器
        public static class UnivariateSelector {
            public List<Feature> select(List<Feature> features, Dataset dataset, int topK) {
                List<FeatureScore> featureScores = new ArrayList<>();

                // 计算每个特征的单变量分数
                for (Feature feature : features) {
                    double score = calculateUnivariateScore(feature, dataset);
                    featureScores.add(new FeatureScore(feature, score));
                }

                // 按分数排序
                featureScores.sort((a, b) -> Double.compare(b.getScore(), a.getScore()));

                // 返回topK特征
                return featureScores.stream()
                    .limit(topK)
                    .map(FeatureScore::getFeature)
                    .collect(Collectors.toList());
            }

            private double calculateUnivariateScore(Feature feature, Dataset dataset) {
                List<Double> featureValues = dataset.getColumn(feature.getName()).stream()
                    .filter(Objects::nonNull)
                    .map(v -> ((Number) v).doubleValue())
                    .collect(Collectors.toList());

                List<Double> targetValues = dataset.getColumn("target").stream()
                    .filter(Objects::nonNull)
                    .map(v -> ((Number) v).doubleValue())
                    .collect(Collectors.toList());

                // 计算相关系数作为分数
                return calculateCorrelation(featureValues, targetValues);
            }

            private double calculateCorrelation(List<Double> x, List<Double> y) {
                int n = Math.min(x.size(), y.size());
                if (n < 2) return 0;

                double sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;

                for (int i = 0; i < n; i++) {
                    sumX += x.get(i);
                    sumY += y.get(i);
                    sumXY += x.get(i) * y.get(i);
                    sumX2 += x.get(i) * x.get(i);
                    sumY2 += y.get(i) * y.get(i);
                }

                double numerator = n * sumXY - sumX * sumY;
                double denominator = Math.sqrt((n * sumX2 - sumX * sumX) * (n * sumY2 - sumY * sumY));

                return denominator == 0 ? 0 : numerator / denominator;
            }

            private static class FeatureScore {
                private final Feature feature;
                private final double score;

                public FeatureScore(Feature feature, double score) {
                    this.feature = feature;
                    this.score = score;
                }

                public Feature getFeature() { return feature; }
                public double getScore() { return score; }
            }
        }

        // 基于模型的选择器
        public static class ModelBasedSelector {
            public List<Feature> select(List<Feature> features, Dataset dataset, int topK) {
                // 使用随机森林进行特征重要性评估
                List<FeatureImportance> importances = evaluateFeatureImportance(features, dataset);

                // 按重要性排序
                importances.sort((a, b) -> Double.compare(b.getImportance(), a.getImportance()));

                // 返回topK特征
                return importances.stream()
                    .limit(topK)
                    .map(FeatureImportance::getFeature)
                    .collect(Collectors.toList());
            }

            private List<FeatureImportance> evaluateFeatureImportance(List<Feature> features, Dataset dataset) {
                List<FeatureImportance> importances = new ArrayList<>();
                Random random = new Random(42);

                for (Feature feature : features) {
                    // 模拟特征重要性计算
                    double importance = 0.1 + random.nextDouble() * 0.9;
                    importances.add(new FeatureImportance(feature, importance));
                }

                return importances;
            }

            private static class FeatureImportance {
                private final Feature feature;
                private final double importance;

                public FeatureImportance(Feature feature, double importance) {
                    this.feature = feature;
                    this.importance = importance;
                }

                public Feature getFeature() { return feature; }
                public double getImportance() { return importance; }
            }
        }

        // 递归特征消除器
        public static class RecursiveFeatureEliminator {
            public List<Feature> select(List<Feature> features, Dataset dataset, int topK) {
                // 简化实现：随机选择topK特征
                List<Feature> shuffled = new ArrayList<>(features);
                Collections.shuffle(shuffled, new Random(42));
                return shuffled.stream().limit(topK).collect(Collectors.toList());
            }
        }
    }

    public static class StatisticalFeatureExtractor implements FeatureExtractor {
        @Override
        public List<Feature> extract(Dataset dataset) {
            // 统计特征提取实现
            return new ArrayList<>();
        }
    }

    public static class TemporalFeatureExtractor implements FeatureExtractor {
        @Override
        public List<Feature> extract(Dataset dataset) {
            // 时序特征提取实现
            List<Feature> features = new ArrayList<>();

            // 查找时间戳列
            Optional<String> timestampColumn = dataset.getColumnNames().stream()
                .filter(name -> name.toLowerCase().contains("time") || name.toLowerCase().contains("date"))
                .findFirst();

            if (timestampColumn.isPresent()) {
                List<Object> timestamps = dataset.getColumn(timestampColumn.get());
                features.addAll(extractTimeBasedFeatures(timestamps));
            }

            return features;
        }

        private List<Feature> extractTimeBasedFeatures(List<Object> timestamps) {
            List<Feature> features = new ArrayList<>();

            // 提取时间特征：小时、星期几、月份等
            for (int i = 0; i < timestamps.size(); i++) {
                Object ts = timestamps.get(i);
                if (ts instanceof Long) {
                    Instant instant = Instant.ofEpochMilli((Long) ts);
                    LocalDateTime dateTime = LocalDateTime.ofInstant(instant, ZoneId.systemDefault());

                    features.add(new Feature("hour_" + i, dateTime.getHour(), FeatureType.TEMPORAL));
                    features.add(new Feature("day_of_week_" + i, dateTime.getDayOfWeek().getValue(), FeatureType.TEMPORAL));
                    features.add(new Feature("month_" + i, dateTime.getMonthValue(), FeatureType.TEMPORAL));
                }
            }

            return features;
        }
    }

    public static class InteractionFeatureExtractor implements FeatureExtractor {
        @Override
        public List<Feature> extract(Dataset dataset) {
            List<Feature> features = new ArrayList<>();
            List<String> numericColumns = dataset.getColumnNames().stream()
                .filter(col -> isNumericColumn(dataset.getColumn(col)))
                .collect(Collectors.toList());

            // 生成交叉特征
            for (int i = 0; i < numericColumns.size(); i++) {
                for (int j = i + 1; j < numericColumns.size(); j++) {
                    String col1 = numericColumns.get(i);
                    String col2 = numericColumns.get(j);

                    // 乘积特征
                    features.add(createInteractionFeature(dataset, col1, col2, "*", (a, b) -> a * b));

                    // 比值特征
                    features.add(createInteractionFeature(dataset, col1, col2, "ratio", (a, b) -> b != 0 ? a / b : 0));

                    // 差值特征
                    features.add(createInteractionFeature(dataset, col1, col2, "diff", (a, b) -> a - b));
                }
            }

            return features;
        }

        private boolean isNumericColumn(List<Object> column) {
            return column.stream().filter(Objects::nonNull).allMatch(v -> v instanceof Number);
        }

        private Feature createInteractionFeature(Dataset dataset, String col1, String col2,
                                               String suffix, BinaryOperator<Double> operation) {
            List<Object> values1 = dataset.getColumn(col1);
            List<Object> values2 = dataset.getColumn(col2);
            List<Double> interactionValues = new ArrayList<>();

            for (int i = 0; i < Math.min(values1.size(), values2.size()); i++) {
                Object v1 = values1.get(i);
                Object v2 = values2.get(i);

                if (v1 instanceof Number && v2 instanceof Number) {
                    double val1 = ((Number) v1).doubleValue();
                    double val2 = ((Number) v2).doubleValue();
                    interactionValues.add(operation.apply(val1, val2));
                } else {
                    interactionValues.add(null);
                }
            }

            String featureName = col1 + "_" + col2 + "_" + suffix;
            return new Feature(featureName, interactionValues, FeatureType.INTERACTION);
        }

        @FunctionalInterface
        private interface BinaryOperator<T> {
            T apply(T a, T b);
        }
    }

    public static class PolynomialFeatureExtractor implements FeatureExtractor {
        @Override
        public List<Feature> extract(Dataset dataset) {
            List<Feature> features = new ArrayList<>();

            for (String column : dataset.getColumnNames()) {
                if (isNumericColumn(dataset.getColumn(column))) {
                    // 二次项特征
                    features.add(createPolynomialFeature(dataset, column, 2));
                    // 三次项特征
                    features.add(createPolynomialFeature(dataset, column, 3));
                }
            }

            return features;
        }

        private boolean isNumericColumn(List<Object> column) {
            return column.stream().filter(Objects::nonNull).allMatch(v -> v instanceof Number);
        }

        private Feature createPolynomialFeature(Dataset dataset, String column, int degree) {
            List<Object> originalValues = dataset.getColumn(column);
            List<Double> polyValues = new ArrayList<>();

            for (Object value : originalValues) {
                if (value instanceof Number) {
                    double val = ((Number) value).doubleValue();
                    polyValues.add(Math.pow(val, degree));
                } else {
                    polyValues.add(null);
                }
            }

            String featureName = column + "_poly_" + degree;
            return new Feature(featureName, polyValues, FeatureType.POLYNOMIAL);
        }
    }

    public static class TextFeatureExtractor implements FeatureExtractor {
        @Override
        public List<Feature> extract(Dataset dataset) {
            List<Feature> features = new ArrayList<>();

            for (String column : dataset.getColumnNames()) {
                if (isTextColumn(dataset.getColumn(column))) {
                    features.addAll(extractTextFeaturesForColumn(dataset, column));
                }
            }

            return features;
        }

        private boolean isTextColumn(List<Object> column) {
            return column.stream()
                .filter(Objects::nonNull)
                .allMatch(v -> v instanceof String);
        }

        private List<Feature> extractTextFeaturesForColumn(Dataset dataset, String column) {
            List<Feature> features = new ArrayList<>();
            List<Object> textValues = dataset.getColumn(column);

            for (int i = 0; i < textValues.size(); i++) {
                String text = (String) textValues.get(i);
                if (text != null) {
                    // 文本长度
                    features.add(new Feature(column + "_length_" + i, text.length(), FeatureType.TEXT));

                    // 单词数量
                    int wordCount = text.split("\\s+").length;
                    features.add(new Feature(column + "_word_count_" + i, wordCount, FeatureType.TEXT));

                    // 大写字母比例
                    long upperCaseCount = text.chars().filter(Character::isUpperCase).count();
                    double upperCaseRatio = text.length() > 0 ? (double) upperCaseCount / text.length() : 0;
                    features.add(new Feature(column + "_upper_ratio_" + i, upperCaseRatio, FeatureType.TEXT));
                }
            }

            return features;
        }
    }

    public static class CategoryFeatureExtractor implements FeatureExtractor {
        @Override
        public List<Feature> extract(Dataset dataset) {
            List<Feature> features = new ArrayList<>();

            for (String column : dataset.getColumnNames()) {
                if (isCategoryColumn(dataset.getColumn(column))) {
                    // One-Hot编码
                    features.addAll(createOneHotFeatures(dataset, column));
                }
            }

            return features;
        }

        private boolean isCategoryColumn(List<Object> column) {
            long uniqueCount = column.stream().filter(Objects::nonNull).distinct().count();
            long totalCount = column.stream().filter(Objects::nonNull).count();
            return uniqueCount < totalCount * 0.5 && uniqueCount < 50; // 假设分类字段唯一值较少
        }

        private List<Feature> createOneHotFeatures(Dataset dataset, String column) {
            List<Feature> features = new ArrayList<>();
            Set<Object> uniqueValues = dataset.getColumn(column).stream()
                .filter(Objects::nonNull)
                .collect(Collectors.toSet());

            for (Object value : uniqueValues) {
                List<Integer> oneHotValues = new ArrayList<>();
                for (Object rowValue : dataset.getColumn(column)) {
                    oneHotValues.add(Objects.equals(rowValue, value) ? 1 : 0);
                }

                String featureName = column + "_" + value + "_onehot";
                features.add(new Feature(featureName, oneHotValues.stream()
                    .map(Integer::doubleValue).collect(Collectors.toList()), FeatureType.CATEGORY));
            }

            return features;
        }
    }
