# AI公平性与算法偏见检测

## 🎯 面试题目

### 题目1：算法公平性评估与偏见检测 ⭐⭐⭐⭐
**问题**：设计并实现一个完整的AI公平性评估系统，能够检测和缓解机器学习模型中的偏见，包括统计公平性度量、因果公平性分析和偏见缓解算法。

**答案要点**：
1. 公平性的不同定义和度量方法
2. 偏见的来源和检测技术
3. 偏见缓解策略和实现方法

**核心原理**：
AI公平性主要关注以下几个方面：
- **群体公平性**：不同群体应获得相似的待遇
- **个体公平性**：相似的个体应获得相似的待遇
- **反事实公平性**：在改变敏感属性后，预测结果应该保持不变

```java
import java.util.*;
import java.util.stream.Collectors;

/**

public class FairnessAnalyzer {

    // 公平性度量类
    public static class FairnessMetrics {

        // 统计奇偶性（Demographic Parity）
        public static double demographicParity(List<Prediction> predictions, String sensitiveAttribute) {
            Map<Object, Double> groupRates = new HashMap<>();

            for (String group : getUniqueGroups(predictions, sensitiveAttribute)) {
                List<Prediction> groupPredictions = getGroupPredictions(predictions, sensitiveAttribute, group);
                double positiveRate = groupPredictions.stream()
                    .mapToDouble(p -> p.predictedLabel)
                    .average()
                    .orElse(0.0);
                groupRates.put(group, positiveRate);
            }

            return calculateVariance(groupRates.values());
        }

        // 均等化奇率（Equalized Odds）
        public static double equalizedOdds(List<Prediction> predictions, String sensitiveAttribute) {
            Map<Object, Double> groupTPRs = new HashMap<>();
            Map<Object, Double> groupFPRs = new HashMap<>();

            for (String group : getUniqueGroups(predictions, sensitiveAttribute)) {
                List<Prediction> groupPredictions = getGroupPredictions(predictions, sensitiveAttribute, group);

                // 计算真阳性率和假阳性率
                ConfusionMatrix cm = calculateConfusionMatrix(groupPredictions);
                groupTPRs.put(group, cm.truePositiveRate());
                groupFPRs.put(group, cm.falsePositiveRate());
            }

            double tprVariance = calculateVariance(groupTPRs.values());
            double fprVariance = calculateVariance(groupFPRs.values());

            return (tprVariance + fprVariance) / 2;
        }

        // 机会均等（Equal Opportunity）
        public static double equalOpportunity(List<Prediction> predictions, String sensitiveAttribute) {
            Map<Object, Double> groupTPRs = new HashMap<>();

            for (String group : getUniqueGroups(predictions, sensitiveAttribute)) {
                List<Prediction> groupPredictions = getGroupPredictions(predictions, sensitiveAttribute, group);
                ConfusionMatrix cm = calculateConfusionMatrix(groupPredictions);
                groupTPRs.put(group, cm.truePositiveRate());
            }

            return calculateVariance(groupTPRs.values());
        }

        // 预测精度均等（Predictive Parity）
        public static double predictiveParity(List<Prediction> predictions, String sensitiveAttribute) {
            Map<Object, Double> groupPPVs = new HashMap<>();

            for (String group : getUniqueGroups(predictions, sensitiveAttribute)) {
                List<Prediction> groupPredictions = getGroupPredictions(predictions, sensitiveAttribute, group);
                ConfusionMatrix cm = calculateConfusionMatrix(groupPredictions);
                groupPPVs.put(group, cm.precision());
            }

            return calculateVariance(groupPPVs.values());
        }

        // 辅助方法
        private static Set<String> getUniqueGroups(List<Prediction> predictions, String sensitiveAttribute) {
            return predictions.stream()
                .map(p -> p.getAttributes().get(sensitiveAttribute).toString())
                .collect(Collectors.toSet());
        }

        private static List<Prediction> getGroupPredictions(List<Prediction> predictions,
                                                           String sensitiveAttribute, String group) {
            return predictions.stream()
                .filter(p -> p.getAttributes().get(sensitiveAttribute).toString().equals(group))
                .collect(Collectors.toList());
        }

        private static double calculateVariance(Collection<Double> values) {
            double mean = values.stream().mapToDouble(Double::doubleValue).average().orElse(0);
            double variance = values.stream()
                .mapToDouble(v -> Math.pow(v - mean, 2))
                .average()
                .orElse(0);
            return variance;
        }

        private static ConfusionMatrix calculateConfusionMatrix(List<Prediction> predictions) {
            int tp = 0, tn = 0, fp = 0, fn = 0;

            for (Prediction p : predictions) {
                if (p.predictedLabel == 1 && p.trueLabel == 1) tp++;
                else if (p.predictedLabel == 0 && p.trueLabel == 0) tn++;
                else if (p.predictedLabel == 1 && p.trueLabel == 0) fp++;
                else if (p.predictedLabel == 0 && p.trueLabel == 1) fn++;
            }

            return new ConfusionMatrix(tp, tn, fp, fn);
        }
    }

    // 混淆矩阵类
    public static class ConfusionMatrix {
        public final int truePositives;
        public final int trueNegatives;
        public final int falsePositives;
        public final int falseNegatives;

        public ConfusionMatrix(int tp, int tn, int fp, int fn) {
            this.truePositives = tp;
            this.trueNegatives = tn;
            this.falsePositives = fp;
            this.falseNegatives = fn;
        }

        public double truePositiveRate() {
            int denominator = truePositives + falseNegatives;
            return denominator == 0 ? 0 : (double) truePositives / denominator;
        }

        public double falsePositiveRate() {
            int denominator = falsePositives + trueNegatives;
            return denominator == 0 ? 0 : (double) falsePositives / denominator;
        }

        public double precision() {
            int denominator = truePositives + falsePositives;
            return denominator == 0 ? 0 : (double) truePositives / denominator;
        }

        public double recall() {
            return truePositiveRate();
        }

        public double accuracy() {
            int total = truePositives + trueNegatives + falsePositives + falseNegatives;
            return total == 0 ? 0 : (double) (truePositives + trueNegatives) / total;
        }
    }

    // 偏见检测器
    public static class BiasDetector {

        // 检测数据集偏见
        public static BiasReport detectDatasetBias(List<DataInstance> dataset, String sensitiveAttribute) {
            BiasReport report = new BiasReport();

            // 计算基线统计
            Map<Object, Integer> groupCounts = new HashMap<>();
            Map<Object, Double> groupOutcomes = new HashMap<>();

            for (DataInstance instance : dataset) {
                Object group = instance.getAttributes().get(sensitiveAttribute);
                groupCounts.put(group, groupCounts.getOrDefault(group, 0) + 1);

                if (instance.getLabel() != null) {
                    double outcome = Double.parseDouble(instance.getLabel().toString());
                    groupOutcomes.merge(group, outcome, Double::sum);
                }
            }

            // 计算总体统计
            int totalInstances = dataset.size();
            for (Map.Entry<Object, Integer> entry : groupCounts.entrySet()) {
                Object group = entry.getKey();
                int count = entry.getValue();
                double proportion = (double) count / totalInstances;
                double avgOutcome = groupOutcomes.getOrDefault(group, 0.0) / count;

                report.addGroupStatistics(group.toString(), count, proportion, avgOutcome);
            }

            // 检测代表性偏见
            double representationVariance = calculateRepresentationBias(groupCounts, totalInstances);
            report.setRepresentationBias(representationVariance);

            // 检测标签偏见
            double labelBias = calculateLabelBias(groupOutcomes, groupCounts);
            report.setLabelBias(labelBias);

            return report;
        }

        // 检测模型偏见
        public static BiasReport detectModelBias(List<Prediction> predictions, String sensitiveAttribute) {
            BiasReport report = new BiasReport();

            // 计算不同公平性度量
            double demoParity = FairnessMetrics.demographicParity(predictions, sensitiveAttribute);
            double equalOdds = FairnessMetrics.equalizedOdds(predictions, sensitiveAttribute);
            double equalOpp = FairnessMetrics.equalOpportunity(predictions, sensitiveAttribute);
            double predParity = FairnessMetrics.predictiveParity(predictions, sensitiveAttribute);

            report.setDemographicParity(demoParity);
            report.setEqualizedOdds(equalOdds);
            report.setEqualOpportunity(equalOpp);
            report.setPredictiveParity(predParity);

            // 计算总体偏见分数
            double overallBias = (demoParity + equalOdds + equalOpp + predParity) / 4;
            report.setOverallBias(overallBias);

            // 生成偏见等级
            if (overallBias < 0.01) {
                report.setBiasLevel("低");
            } else if (overallBias < 0.05) {
                report.setBiasLevel("中");
            } else if (overallBias < 0.1) {
                report.setBiasLevel("高");
            } else {
                report.setBiasLevel("严重");
            }

            return report;
        }

        private static double calculateRepresentationBias(Map<Object, Integer> groupCounts, int total) {
            double expectedProportion = 1.0 / groupCounts.size();
            double variance = 0;

            for (int count : groupCounts.values()) {
                double actualProportion = (double) count / total;
                variance += Math.pow(actualProportion - expectedProportion, 2);
            }

            return variance / groupCounts.size();
        }

        private static double calculateLabelBias(Map<Object, Double> groupOutcomes, Map<Object, Integer> groupCounts) {
            double overallMean = groupOutcomes.values().stream()
                .mapToDouble(Double::doubleValue)
                .sum() / groupCounts.values().stream().mapToDouble(Integer::intValue).sum();

            double variance = 0;
            for (Map.Entry<Object, Double> entry : groupOutcomes.entrySet()) {
                Object group = entry.getKey();
                double groupMean = entry.getValue() / groupCounts.get(group);
                variance += Math.pow(groupMean - overallMean, 2);
            }

            return variance / groupCounts.size();
        }
    }

    // 偏见缓解算法
    public static class BiasMitigation {

        // 预处理：重新加权
        public static List<DataInstance> reweighing(List<DataInstance> dataset,
                                                  String sensitiveAttribute, String labelAttribute) {
            // 计算组权重
            Map<Object, Double> groupWeights = calculateGroupWeights(dataset, sensitiveAttribute, labelAttribute);

            // 应用权重
            List<DataInstance> reweightedDataset = new ArrayList<>();
            for (DataInstance instance : dataset) {
                Object group = instance.getAttributes().get(sensitiveAttribute);
                double weight = groupWeights.get(group);
                reweightedDataset.add(new DataInstance(
                    instance.getAttributes(),
                    instance.getLabel(),
                    weight
                ));
            }

            return reweightedDataset;
        }

        // 预处理：优化预处理
        public static List<DataInstance> optimizedPreprocessing(List<DataInstance> dataset,
                                                               String sensitiveAttribute,
                                                               String labelAttribute) {
            // 优化数据分布以实现统计奇偶性
            Map<Object, List<DataInstance>> groupedData = dataset.stream()
                .collect(Collectors.groupingBy(d -> d.getAttributes().get(sensitiveAttribute)));

            // 计算目标分布
            int targetSize = groupedData.values().stream()
                .mapToInt(List::size)
                .max()
                .orElse(0);

            List<DataInstance> balancedDataset = new ArrayList<>();

            for (Map.Entry<Object, List<DataInstance>> entry : groupedData.entrySet()) {
                List<DataInstance> groupData = entry.getValue();

                // 平衡样本数量
                if (groupData.size() < targetSize) {
                    // 过采样
                    balancedDataset.addAll(oversample(groupData, targetSize));
                } else {
                    // 欠采样
                    balancedDataset.addAll(undersample(groupData, targetSize));
                }
            }

            return balancedDataset;
        }

        // 后处理：均衡后处理
        public static List<Prediction> equalizedOddsPostprocessing(List<Prediction> predictions,
                                                                 String sensitiveAttribute) {
            Map<Object, List<Prediction>> groupedPredictions = predictions.stream()
                .collect(Collectors.groupingBy(p -> p.getAttributes().get(sensitiveAttribute)));

            // 为每个组找到最优阈值
            Map<Object, Double> groupThresholds = new HashMap<>();
            for (Map.Entry<Object, List<Prediction>> entry : groupedPredictions.entrySet()) {
                Object group = entry.getKey();
                List<Prediction> groupPreds = entry.getValue();
                double threshold = findOptimalThreshold(groupPreds);
                groupThresholds.put(group, threshold);
            }

            // 应用组特定阈值
            List<Prediction> adjustedPredictions = new ArrayList<>();
            for (Prediction pred : predictions) {
                Object group = pred.getAttributes().get(sensitiveAttribute);
                double threshold = groupThresholds.get(group);
                double adjustedScore = pred.predictedLabel;

                // 调整预测分数以实现均衡化奇率
                if (adjustedScore >= threshold) {
                    adjustedScore = 1.0;
                } else {
                    adjustedScore = 0.0;
                }

                adjustedPredictions.add(new Prediction(
                    pred.getAttributes(),
                    pred.trueLabel,
                    adjustedScore,
                    null
                ));
            }

            return adjustedPredictions;
        }

        private static Map<Object, Double> calculateGroupWeights(List<DataInstance> dataset,
                                                               String sensitiveAttribute,
                                                               String labelAttribute) {
            Map<Object, Map<Object, Integer>> groupLabelCounts = new HashMap<>();
            Map<Object, Integer> labelCounts = new HashMap<>();
            int total = dataset.size();

            // 计算组-标签计数
            for (DataInstance instance : dataset) {
                Object group = instance.getAttributes().get(sensitiveAttribute);
                Object label = instance.getLabel();

                groupLabelCounts.computeIfAbsent(group, k -> new HashMap<>())
                    .merge(label, 1, Integer::sum);
                labelCounts.merge(label, 1, Integer::sum);
            }

            // 计算权重
            Map<Object, Double> weights = new HashMap<>();
            for (Map.Entry<Object, Map<Object, Integer>> groupEntry : groupLabelCounts.entrySet()) {
                Object group = groupEntry.getKey();
                double groupWeight = 0;

                for (Map.Entry<Object, Integer> labelEntry : groupEntry.getValue().getBytes()) {
                    Object label = labelEntry.getKey();
                    int groupLabelCount = labelEntry.getValue();
                    int labelCount = labelCounts.get(label);

                    double expected = (groupEntry.getValue().values().stream().mapToInt(Integer::intValue).sum() *
                                     (double) labelCount) / total;
                    double weight = expected / groupLabelCount;
                    groupWeight += weight * groupLabelCount;
                }

                weights.put(group, groupWeight / groupEntry.getValue().values().stream().mapToInt(Integer::intValue).sum());
            }

            return weights;
        }

        private static List<DataInstance> oversample(List<DataInstance> data, int targetSize) {
            List<DataInstance> oversampled = new ArrayList<>(data);
            Random random = new Random();

            while (oversampled.size() < targetSize) {
                DataInstance sample = data.get(random.nextInt(data.size()));
                oversampled.add(sample);
            }

            return oversampled;
        }

        private static List<DataInstance> undersample(List<DataInstance> data, int targetSize) {
            Random random = new Random();
            List<DataInstance> undersampled = new ArrayList<>(data);
            Collections.shuffle(undersampled, random);
            return undersampled.subList(0, Math.min(targetSize, undersampled.size()));
        }

        private static double findOptimalThreshold(List<Prediction> predictions) {
            // 简化的阈值优化
            // 实际实现应该优化特定的公平性度量
            return 0.5;
        }
    }

    public static class DataInstance {
        private final Map<String, Object> attributes;
        private final Object label;
        private final double weight;

        public DataInstance(Map<String, Object> attributes, Object label) {
            this(attributes, label, 1.0);
        }

        public DataInstance(Map<String, Object> attributes, Object label, double weight) {
            this.attributes = new HashMap<>(attributes);
            this.label = label;
            this.weight = weight;
        }

        public Map<String, Object> getAttributes() {
            return Collections.unmodifiableMap(attributes);
        }

        public Object getLabel() {
            return label;
        }

        public double getWeight() {
            return weight;
        }

    public static class BiasReport {
        private double demographicParity;
        private double equalizedOdds;
        private double equalOpportunity;
        private double predictiveParity;
        private double overallBias;
        private String biasLevel;
        private double representationBias;
        private double labelBias;

        private final Map<String, GroupStatistics> groupStatistics = new HashMap<>();

        // Getters and setters

        public void setPredictiveParity(double predictiveParity) { this.predictiveParity = predictiveParity; }

        public double getOverallBias() { return overallBias; }
        public void setOverallBias(double overallBias) { this.overallBias = overallBias; }

        public String getBiasLevel() { return biasLevel; }
        public void setBiasLevel(String biasLevel) { this.biasLevel = biasLevel; }

        public double getRepresentationBias() { return representationBias; }

        public double getLabelBias() { return labelBias; }

        public Map<String, GroupStatistics> getGroupStatistics() {
            return Collections.unmodifiableMap(groupStatistics);
        }

        @Override
        public String toString() {
            return String.format(
                "BiasReport{\n" +
                "  Demographic Parity: %.4f\n" +
                "  Equalized Odds: %.4f\n" +
                "  Equal Opportunity: %.4f\n" +
                "  Predictive Parity: %.4f\n" +
                "  Overall Bias: %.4f (%s)\n" +
                "  Representation Bias: %.4f\n" +
                "  Label Bias: %.4f\n" +
                "  Group Statistics: %s\n" +

    private static List<DataInstance> createSampleDataset() {
        List<DataInstance> dataset = new ArrayList<>();
        Random random = new Random();

        // 创建有偏见的数据集
        for (int i = 0; i < 1000; i++) {
            Map<String, Object> attributes = new HashMap<>();

            // 敏感属性：性别
            String gender = random.nextDouble() < 0.6 ? "male" : "female"; // 不平衡的性别分布
            attributes.put("gender", gender);

            // 其他特征
            attributes.put("age", 20 + random.nextInt(50));
            attributes.put("income", 30000 + random.nextInt(70000));
            attributes.put("education", random.nextInt(5));

            // 有偏见的标签分配
            double baseProb = gender.equals("male") ? 0.7 : 0.3; // 性别偏见
            double prob = baseProb + (random.nextGaussian() * 0.1);
            int label = random.nextDouble() < prob ? 1 : 0;

            dataset.add(new DataInstance(attributes, label));
        }

        return dataset;
    }

    private static List<Prediction> createSamplePredictions() {
        List<Prediction> predictions = new ArrayList<>();
        Random random = new Random();

        for (int i = 0; i < 1000; i++) {
            Map<String, Object> attributes = new HashMap<>();
            String gender = random.nextDouble() < 0.6 ? "male" : "female";
            attributes.put("gender", gender);

            // 有偏见的预测
            double baseProb = gender.equals("male") ? 0.75 : 0.25; // 模型偏见
            double predictionProb = Math.max(0, Math.min(1, baseProb + random.nextGaussian() * 0.1));
            double predictedLabel = random.nextDouble() < predictionProb ? 1 : 0;

            // 真实标签（也可能有偏见）
            double trueProb = gender.equals("male") ? 0.7 : 0.3;
            double trueLabel = random.nextDouble() < trueProb ? 1 : 0;

            predictions.add(new Prediction(attributes, trueLabel, predictedLabel, predictionProb));
        }

        return predictions;
    }
```

### 题目2：可解释性AI（XAI）算法实现 ⭐⭐⭐⭐⭐
**问题**：实现LIME（Local Interpretable Model-agnostic Explanations）和SHAP（SHapley Additive exPlanations）算法，为机器学习模型提供局部和全局解释。

**答案要点**：
1. 局部可解释性的重要性
2. 模型无关的解释方法
3. 特征重要性的计算和可视化

**核心代码示例**：

```java
import java.util.*;
import java.util.stream.Collectors;

/**
 * 可解释性AI（XAI）算法实现
 */
public class ExplainableAI {

    // LIME算法实现
    public static class LIMEExplainer {
        private final PredictiveModel model;
        private final int numSamples;
        private final double kernelWidth;
        private final Random random;

        public LIMEExplainer(PredictiveModel model, int numSamples, double kernelWidth) {
            this.model = model;
            this.numSamples = numSamples;
            this.kernelWidth = kernelWidth;
            this.random = new Random();
        }

        // 生成局部解释
        public LimeExplanation explainLocal(Map<String, Double> instance) {
            // 1. 生成扰动样本
            List<PerturbedSample> perturbedSamples = generatePerturbedSamples(instance);

            // 2. 计算样本权重（距离权重）
            for (PerturbedSample sample : perturbedSamples) {
                double distance = calculateDistance(instance, sample.getFeatures());
                sample.setWeight(calculateWeight(distance));
            }

            // 3. 拟合局部可解释模型
            LinearSurrogateModel surrogateModel = fitSurrogateModel(perturbedSamples);

            // 4. 生成解释
            return new LimeExplanation(instance, surrogateModel, perturbedSamples);
        }

        private List<PerturbedSample> generatePerturbedSamples(Map<String, Double> instance) {
            List<PerturbedSample> samples = new ArrayList<>();
            Set<String> featureNames = instance.keySet();

            // 生成原始样本
            double originalPrediction = model.predict(instance);
            samples.add(new PerturbedSample(
                new HashMap<>(instance),
                originalPrediction,
                1.0
            ));

            // 生成扰动样本
            for (int i = 0; i < numSamples - 1; i++) {
                Map<String, Double> perturbedFeatures = new HashMap<>();
                Set<String> selectedFeatures = new HashSet<>();

                // 随机选择特征子集
                for (String feature : featureNames) {
                    if (random.nextDouble() < 0.5) {
                        selectedFeatures.add(feature);
                        perturbedFeatures.put(feature, instance.get(feature));
                    } else {
                        // 用随机值或平均值替换
                        perturbedFeatures.put(feature, getRandomFeatureValue(feature));
                    }
                }

                double prediction = model.predict(perturbedFeatures);
                samples.add(new PerturbedSample(perturbedFeatures, prediction, 0));
            }

            return samples;
        }

        private double calculateDistance(Map<String, Double> original, Map<String, Double> perturbed) {
            double distance = 0;
            for (String feature : original.keySet()) {
                double diff = original.get(feature) - perturbed.get(feature);
                distance += diff * diff;
            }
            return Math.sqrt(distance);
        }

        private double calculateWeight(double distance) {
            return Math.exp(-distance * distance / (kernelWidth * kernelWidth));
        }

        private LinearSurrogateModel fitSurrogateModel(List<PerturbedSample> samples) {
            return new LinearSurrogateModel(samples);
        }

        private Double getRandomFeatureValue(String feature) {
            // 简化实现：返回随机值
            // 实际应该基于特征分布或使用全局平均值
            return random.nextDouble() * 10;
        }
    }

    // SHAP算法实现
    public static class SHAPExplainer {
        private final PredictiveModel model;
        private final Random random;

        public SHAPExplainer(PredictiveModel model) {
            this.model = model;
            this.random = new Random();
        }

        // 计算SHAP值
        public ShapExplanation explain(Map<String, Double> instance, List<Map<String, Double>> backgroundDataset) {
            Map<String, Double> shapValues = new HashMap<>();
            Set<String> featureNames = instance.keySet();

            // 计算每个特征的SHAP值
            for (String feature : featureNames) {
                double shapValue = calculateFeatureShapValue(instance, feature, backgroundDataset);
                shapValues.put(feature, shapValue);
            }

            return new ShapExplanation(instance, shapValues, backgroundDataset);
        }

        private double calculateFeatureShapValue(Map<String, Double> instance, String feature,
                                                List<Map<String, Double>> backgroundDataset) {
            double marginalContribution = 0;
            int numBackgroundSamples = Math.min(backgroundDataset.size(), 100);

            for (int i = 0; i < numBackgroundSamples; i++) {
                Map<String, Double> backgroundInstance = backgroundDataset.get(random.nextInt(backgroundDataset.size()));

                // 创建背景实例的副本
                Map<String, Double> backgroundCopy = new HashMap<>(backgroundInstance);

                // 计算有特征时的预测
                Map<String, Double> withFeature = new HashMap<>(backgroundCopy);
                withFeature.put(feature, instance.get(feature));
                double predictionWithFeature = model.predict(withFeature);

                // 计算无特征时的预测
                Map<String, Double> withoutFeature = new HashMap<>(backgroundCopy);
                double predictionWithoutFeature = model.predict(withoutFeature);

                // 计算边际贡献
                marginalContribution += predictionWithFeature - predictionWithoutFeature;
            }

            return marginalContribution / numBackgroundSamples;
        }

        // 批量计算SHAP值
        public Map<String, Double> explainGlobal(List<Map<String, Double>> dataset,
                                               List<Map<String, Double>> backgroundDataset) {
            Map<String, Double> globalShapValues = new HashMap<>();

            // 计算所有实例的平均SHAP值
            for (Map<String, Double> instance : dataset) {
                ShapExplanation explanation = explain(instance, backgroundDataset);
                Map<String, Double> instanceShapValues = explanation.getShapValues();

                for (Map.Entry<String, Double> entry : instanceShapValues.entrySet()) {
                    globalShapValues.merge(entry.getKey(), entry.getValue(), Double::sum);
                }
            }

            // 计算平均值
            for (Map.Entry<String, Double> entry : globalShapValues.entrySet()) {
                entry.setValue(entry.getValue() / dataset.size());
            }

            return globalShapValues;
        }
    }

    // 线性替代模型
    public static class LinearSurrogateModel {
        private final Map<String, Double> coefficients;
        private final double intercept;
        private final double rSquared;

        public LinearSurrogateModel(List<PerturbedSample> samples) {
            // 简化的线性回归实现
            // 实际实现应该使用更复杂的优化算法
            Map<String, Double> coeffs = new HashMap<>();
            double intercept = 0;

            // 为每个特征计算简单的系数
            if (!samples.isEmpty()) {
                Map<String, Double> featureNames = samples.get(0).getFeatures();

                for (String feature : featureNames.keySet()) {
                    double coefficient = calculateFeatureCoefficient(samples, feature);
                    coeffs.put(feature, coefficient);
                }

                intercept = calculateIntercept(samples, coeffs);
            }

            this.coefficients = coeffs;
            this.intercept = intercept;
            this.rSquared = calculateRSquared(samples);
        }

        private double calculateFeatureCoefficient(List<PerturbedSample> samples, String feature) {
            // 简化的系数计算（相关性）
            double covariance = 0;
            double variance = 0;
            double meanFeature = 0;
            double meanTarget = 0;

            // 计算均值
            double totalWeight = 0;
            for (PerturbedSample sample : samples) {
                double weight = sample.getWeight();
                meanFeature += weight * sample.getFeature(feature);
                meanTarget += weight * sample.getPrediction();
                totalWeight += weight;
            }
            meanFeature /= totalWeight;
            meanTarget /= totalWeight;

            // 计算协方差和方差
            for (PerturbedSample sample : samples) {
                double weight = sample.getWeight();
                double featureValue = sample.getFeature(feature);
                double targetValue = sample.getPrediction();

                covariance += weight * (featureValue - meanFeature) * (targetValue - meanTarget);
                variance += weight * (featureValue - meanFeature) * (featureValue - meanFeature);
            }

            return variance == 0 ? 0 : covariance / variance;
        }

        private double calculateIntercept(List<PerturbedSample> samples, Map<String, Double> coefficients) {
            double intercept = 0;
            double totalWeight = 0;

            for (PerturbedSample sample : samples) {
                double weight = sample.getWeight();
                double prediction = sample.getPrediction();
                double modelPrediction = 0;

                for (Map.Entry<String, Double> coeff : coefficients.entrySet()) {
                    modelPrediction += coeff.getValue() * sample.getFeature(coeff.getKey());
                }

                intercept += weight * (prediction - modelPrediction);
                totalWeight += weight;
            }

            return totalWeight == 0 ? 0 : intercept / totalWeight;
        }

        private double calculateRSquared(List<PerturbedSample> samples) {
            double ssRes = 0;
            double ssTot = 0;
            double meanY = 0;
            double totalWeight = 0;

            // 计算均值
            for (PerturbedSample sample : samples) {
                meanY += sample.getWeight() * sample.getPrediction();
                totalWeight += sample.getWeight();
            }
            meanY /= totalWeight;

            // 计算残差平方和和总平方和
            for (PerturbedSample sample : samples) {
                double weight = sample.getWeight();
                double prediction = sample.getPrediction();
                double modelPrediction = predict(sample.getFeatures());

                ssRes += weight * Math.pow(prediction - modelPrediction, 2);
                ssTot += weight * Math.pow(prediction - meanY, 2);
            }

            return ssTot == 0 ? 0 : 1 - (ssRes / ssTot);
        }

        public double predict(Map<String, Double> features) {
            double prediction = intercept;

            for (Map.Entry<String, Double> coeff : coefficients.entrySet()) {
                String feature = coeff.getKey();
                if (features.containsKey(feature)) {
                    prediction += coeff.getValue() * features.get(feature);
                }
            }

            return prediction;
        }

        public Map<String, Double> getCoefficients() {
            return Collections.unmodifiableMap(coefficients);
        }

        public double getIntercept() {
            return intercept;
        }

        public double getRSquared() {
            return rSquared;
        }
    }

    // 扰动样本类
    public static class PerturbedSample {
        private final Map<String, Double> features;
        private final double prediction;
        private double weight;

        public PerturbedSample(Map<String, Double> features, double prediction, double weight) {
            this.features = new HashMap<>(features);
            this.prediction = prediction;
            this.weight = weight;
        }

        public Map<String, Double> getFeatures() {

        public double getPrediction() {
            return prediction;
        }

        public double getWeight() {
            return weight;
        }

        public void setWeight(double weight) {
            this.weight = weight;
        }

        public Double getFeature(String featureName) {
            return features.get(featureName);
        }
    }

    // LIME解释结果类
    public static class LimeExplanation {
        private final Map<String, Double> instance;
        private final LinearSurrogateModel surrogateModel;
        private final List<PerturbedSample> perturbedSamples;

        public LimeExplanation(Map<String, Double> instance, LinearSurrogateModel surrogateModel,
                              List<PerturbedSample> perturbedSamples) {
            this.instance = new HashMap<>(instance);
            this.surrogateModel = surrogateModel;

        }

        }

        public double getIntercept() {
            return surrogateModel.getIntercept();
        }

        public double getFidelity() {
            return surrogateModel.getRSquared();
        }

        public List<Map.Entry<String, Double>> getTopFeatures(int topN) {
            return getFeatureImportance().entrySet().stream()
                .sorted(Map.Entry.<String, Double>comparingByValue().reversed())

                .collect(Collectors.toList());
        }

        @Override
        public String toString() {

            sb.append("LIME解释结果:\n");
            sb.append(String.format("模型拟合度 (R²): %.3f\n", getFidelity()));
            sb.append("特征重要性 (Top 10):\n");

            int rank = 1;
            for (Map.Entry<String, Double> entry : getTopFeatures(10)) {

        }
    }
```
