### 题目1：贝叶斯网络推理引擎实现 ⭐⭐⭐⭐
**问题**：实现一个完整的贝叶斯网络推理引擎，包括精确推理（变量消除）和近似推理（马尔可夫链蒙特卡洛）算法。

**核心原理**：
贝叶斯网络通过有向无环图表示变量间的条件依赖关系：
```
P(X₁, X₂, ..., Xn) = ∏ P(Xi | Parents(Xi))
```

**核心代码示例**：

```java
import java.util.*;

public class BayesianNetwork {

    // 贝叶斯网络节点类
    public static class Node {
        private final String name;
        private final List<String> parents;
        private final Map<String, Double> cpt; // 条件概率表
        private final Set<String> states;

        public Node(String name, List<String> parents, Set<String> states) {
            this.name = name;
            this.parents = new ArrayList<>(parents);
            this.states = new HashSet<>(states);
            this.cpt = new HashMap<>();
        }

        public void addConditionalProbability(String parentAssignment, String state, double probability) {
            String key = parentAssignment + "|" + state;
            cpt.put(key, probability);
        }

        public double getProbability(String parentAssignment, String state) {
            String key = parentAssignment + "|" + state;
            return cpt.getOrDefault(key, 0.0);
        }

        // Getters
        public String getName() { return name; }
        public List<String> getParents() { return Collections.unmodifiableList(parents); }
        public Set<String> getStates() { return Collections.unmodifiableSet(states); }
    }

    // 精确推理：变量消除算法
    public static class VariableElimination {
        private final Map<String, Node> network;

        public VariableElimination(Map<String, Node> network) {
            this.network = new HashMap<>(network);
        }

        public double query(String queryVar, String queryState, Map<String, String> evidence) {
            // 1. 收集所有因子
            List<Factor> factors = collectFactors();

            // 2. 应用证据
            factors = applyEvidence(factors, evidence);

            // 3. 变量消除
            List<String> eliminationOrder = getEliminationOrder(queryVar, evidence.keySet());
            factors = eliminateVariables(factors, eliminationOrder);

            // 4. 计算最终概率
            return computeFinalProbability(factors, queryVar, queryState);
        }

        private List<Factor> collectFactors() {
            List<Factor> factors = new ArrayList<>();

            for (Node node : network.values()) {
                Factor factor = new Factor(node);
                factors.add(factor);
            }

            return factors;
        }

        private List<Factor> applyEvidence(List<Factor> factors, Map<String, String> evidence) {
            List<Factor> result = new ArrayList<>();

            for (Factor factor : factors) {
                Factor reducedFactor = factor.reduce(evidence);
                if (reducedFactor != null) {
                    result.add(reducedFactor);
                }
            }

            return result;
        }

        private List<String> getEliminationOrder(String queryVar, Set<String> evidenceVars) {
            // 简化的消元顺序：先处理非查询、非证据变量
            List<String> eliminationOrder = new ArrayList<>();

            for (String var : network.keySet()) {
                if (!var.equals(queryVar) && !evidenceVars.contains(var)) {
                    eliminationOrder.add(var);
                }
            }

            return eliminationOrder;
        }

        private List<Factor> eliminateVariables(List<Factor> factors, List<String> eliminationOrder) {
            List<Factor> result = new ArrayList<>(factors);

            for (String var : eliminationOrder) {
                result = eliminateVariable(result, var);
            }

            return result;
        }

        private List<Factor> eliminateVariable(List<Factor> factors, String var) {
            // 找到包含要消除变量的因子
            List<Factor> factorsWithVar = new ArrayList<>();
            List<Factor> factorsWithoutVar = new ArrayList<>();

            for (Factor factor : factors) {
                if (factor.getVariables().contains(var)) {
                    factorsWithVar.add(factor);
                } else {
                    factorsWithoutVar.add(factor);
                }
            }

            // 如果没有因子包含该变量，直接返回
            if (factorsWithVar.isEmpty()) {
                return factors;
            }

            // 乘积所有包含变量的因子
            Factor productFactor = factorsWithVar.get(0);
            for (int i = 1; i < factorsWithVar.size(); i++) {
                productFactor = productFactor.multiply(factorsWithVar.get(i));
            }

            // 消除变量（边缘化）
            Factor marginalizedFactor = productFactor.marginalize(var);

            // 返回新因子列表
            List<Factor> result = new ArrayList<>(factorsWithoutVar);
            result.add(marginalizedFactor);

            return result;
        }

        private double computeFinalProbability(List<Factor> factors, String queryVar, String queryState) {
            // 乘积所有剩余因子
            if (factors.isEmpty()) {
                return 0.0;
            }

            Factor finalFactor = factors.get(0);
            for (int i = 1; i < factors.size(); i++) {
                finalFactor = finalFactor.multiply(factors.get(i));
            }

            // 计算查询变量的概率
            return finalFactor.getProbability(queryVar, queryState);
        }
    }

    // 因子类
    public static class Factor {
        private final Set<String> variables;
        private final Map<Map<String, String>, Double> table;

        public Factor(Node node) {
            this.variables = new LinkedHashSet<>();
            variables.add(node.getName());
            variables.addAll(node.getParents());

            this.table = new HashMap<>();
            generateTable(node);
        }

        private void generateTable(Node node) {
            List<String> parentVars = new ArrayList<>(node.getParents());
            List<String> allVars = new ArrayList<>(parentVars);
            allVars.add(node.getName());

            // 生成所有可能的赋值组合
            List<Map<String, String>> assignments = generateAssignments(allVars, node);

            for (Map<String, String> assignment : assignments) {
                double probability = calculateProbability(node, assignment);
                table.put(assignment, probability);
            }
        }

        private List<Map<String, String>> generateAssignments(List<String> variables, Node node) {
            List<Map<String, String>> assignments = new ArrayList<>();
            generateAssignmentsRecursive(variables, 0, new HashMap<>(), assignments, node);
            return assignments;
        }

        private void generateAssignmentsRecursive(List<String> variables, int index,
                                                Map<String, String> currentAssignment,
                                                List<Map<String, String>> assignments, Node node) {
            if (index >= variables.size()) {
                assignments.put(new HashMap<>(currentAssignment), null);
                return;
            }

            String var = variables.get(index);
            Set<String> states = network.get(var).getStates();

            for (String state : states) {
                currentAssignment.put(var, state);
                generateAssignmentsRecursive(variables, index + 1, currentAssignment, assignments, node);
                currentAssignment.remove(var);
            }
        }

        private double calculateProbability(Node node, Map<String, String> assignment) {
            // 构建父节点赋值字符串
            StringBuilder parentAssignment = new StringBuilder();
            for (int i = 0; i < node.getParents().size(); i++) {
                String parent = node.getParents().get(i);
                if (i > 0) parentAssignment.append(",");
                parentAssignment.append(assignment.get(parent));
            }

            String state = assignment.get(node.getName());
            return node.getProbability(parentAssignment.toString(), state);
        }

        public Factor multiply(Factor other) {
            Set<String> newVariables = new LinkedHashSet<>(this.variables);
            newVariables.addAll(other.variables);

            Factor result = new Factor();
            result.variables = newVariables;
            result.table = new HashMap<>();

            // 生成新因子的所有赋值组合
            List<Map<String, String>> assignments = generateAllAssignments(newVariables);

            for (Map<String, String> assignment : assignments) {
                double prob1 = getTableProbability(assignment);
                double prob2 = other.getTableProbability(assignment);
                result.table.put(assignment, prob1 * prob2);
            }

            return result;
        }

        public Factor marginalize(String var) {
            Set<String> newVariables = new LinkedHashSet<>(this.variables);
            newVariables.remove(var);

            Factor result = new Factor();
            result.variables = newVariables;
            result.table = new HashMap<>();

            // 生成新因子的所有赋值组合
            List<Map<String, String>> assignments = generateAllAssignments(newVariables);

            for (Map<String, String> assignment : assignments) {
                double sum = 0;

                // 对被边缘化的变量的所有可能状态求和
                Node varNode = network.get(var);
                for (String state : varNode.getStates()) {
                    Map<String, String> fullAssignment = new HashMap<>(assignment);
                    fullAssignment.put(var, state);
                    sum += getTableProbability(fullAssignment);
                }

                result.table.put(assignment, sum);
            }

            return result;
        }

        public Factor reduce(Map<String, String> evidence) {
            Factor result = new Factor();
            result.variables = new LinkedHashSet<>(this.variables);
            result.table = new HashMap<>();

            List<Map<String, String>> assignments = generateAllAssignments(this.variables);

            for (Map<String, String> assignment : assignments) {
                boolean consistent = true;

                for (Map.Entry<String, String> evidenceEntry : evidence.entrySet()) {
                    String var = evidenceEntry.getKey();
                    String value = evidenceEntry.getValue();

                    if (this.variables.contains(var) &&
                        !assignment.get(var).equals(value)) {
                        consistent = false;
                        break;
                    }
                }

                if (consistent) {
                    double prob = getTableProbability(assignment);
                    result.table.put(assignment, prob);
                }
            }

            return result.table.isEmpty() ? null : result;
        }

        private List<Map<String, String>> generateAllAssignments(Set<String> variables) {
            List<Map<String, String>> assignments = new ArrayList<>();
            generateAllAssignmentsRecursive(new ArrayList<>(variables), 0, new HashMap<>(), assignments);
            return assignments;
        }

        private void generateAllAssignmentsRecursive(List<String> variables, int index,
                                                   Map<String, String> currentAssignment,
                                                   List<Map<String, String>> assignments) {
            if (index >= variables.size()) {
                assignments.put(new HashMap<>(currentAssignment), null);
                return;
            }

            String var = variables.get(index);
            Set<String> states = network.get(var).getStates();

            for (String state : states) {
                currentAssignment.put(var, state);
                generateAllAssignmentsRecursive(variables, index + 1, currentAssignment, assignments);
                currentAssignment.remove(var);
            }
        }

        private double getTableProbability(Map<String, String> assignment) {
            // 简化的概率查找
            return table.getOrDefault(assignment, 0.0);
        }

        public double getProbability(String var, String state) {
            for (Map.Entry<Map<String, String>, Double> entry : table.entrySet()) {
                if (entry.getKey().getOrDefault(var, "").equals(state)) {
                    return entry.getValue();
                }
            }
            return 0.0;
        }

        // Private constructor for internal use
        private Factor() {}

        public Set<String> getVariables() {
            return Collections.unmodifiableSet(variables);
        }
    }
}

        Map<String, BayesianNetwork.Node> network = new HashMap<>();

        smoking.addConditionalProbability("", "false", 0.7);
        network.put("Smoking", smoking);

        // LungCancer节点 (依赖Smoking)
        BayesianNetwork.Node lungCancer = new BayesianNetwork.Node("LungCancer",
            Arrays.asList("Smoking"), binaryStates);
        lungCancer.addConditionalProbability("true", "true", 0.1);
        lungCancer.addConditionalProbability("true", "false", 0.9);
        lungCancer.addConditionalProbability("false", "true", 0.01);
        lungCancer.addConditionalProbability("false", "false", 0.99);
        network.put("LungCancer", lungCancer);

        // Cough节点 (依赖LungCancer)
        BayesianNetwork.Node cough = new BayesianNetwork.Node("Cough",
            Arrays.asList("LungCancer"), binaryStates);
        cough.addConditionalProbability("true", "true", 0.8);
        cough.addConditionalProbability("true", "false", 0.2);
        cough.addConditionalProbability("false", "true", 0.1);
        cough.addConditionalProbability("false", "false", 0.9);
        network.put("Cough", cough);

        // 执行推理
        BayesianNetwork.VariableElimination ve = new BayesianNetwork.VariableElimination(network);

        // 查询：P(LungCancer=true | Cough=true)
        Map<String, String> evidence = new HashMap<>();
        evidence.put("Cough", "true");

        double probability = ve.query("LungCancer", "true", evidence);
        System.out.printf("P(LungCancer=true | Cough=true) = %.4f%n", probability);

        // 查询：P(Smoking=true | Cough=true)
        probability = ve.query("Smoking", "true", evidence);
        System.out.printf("P(Smoking=true | Cough=true) = %.4f%n", probability);
    }
}
```

### 题目2：马尔可夫链蒙特卡洛（MCMC）⭐⭐⭐⭐⭐
**问题**：实现MCMC采样算法，包括Metropolis-Hastings和Gibbs采样，用于复杂概率分布的近似推理。

**核心代码示例**：

```java
import java.util.*;

public class MCMCSampling {

    // Metropolis-Hastings采样器
    public static class MetropolisHastingsSampler {
        private final ProposalDistribution proposal;
        private final TargetDistribution target;
        private final Random random;

        public MetropolisHastingsSampler(ProposalDistribution proposal, TargetDistribution target) {
            this.proposal = proposal;
            this.target = target;
            this.random = new Random();
        }

        public List<Double> sample(int numSamples, double initialState, int burnIn) {
            List<Double> samples = new ArrayList<>();
            double currentState = initialState;

            // 预烧期
            for (int i = 0; i < burnIn; i++) {
                currentState = metropolisStep(currentState);
            }

            // 采样期
            for (int i = 0; i < numSamples; i++) {
                currentState = metropolisStep(currentState);
                samples.add(currentState);
            }

            return samples;
        }

        private double metropolisStep(double currentState) {
            // 提议新状态
            double proposedState = proposal.sample(currentState);

            // 计算接受概率
            double acceptanceProb = calculateAcceptanceProbability(currentState, proposedState);

            // 决定是否接受
            if (random.nextDouble() < acceptanceProb) {
                return proposedState;
            } else {
                return currentState;
            }
        }

        private double calculateAcceptanceProbability(double currentState, double proposedState) {
            double targetCurrent = target.logDensity(currentState);
            double targetProposed = target.logDensity(proposedState);

            double proposalCurrent = proposal.logDensity(currentState, proposedState);
            double proposalProposed = proposal.logDensity(proposedState, currentState);

            double logAcceptanceProb = (targetProposed + proposalProposed) - (targetCurrent + proposalCurrent);

            return Math.min(1.0, Math.exp(logAcceptanceProb));
        }
    }

    // Gibbs采样器
    public static class GibbsSampler {
        private final JointDistribution jointDist;
        private final List<String> variables;
        private final Random random;

        public GibbsSampler(JointDistribution jointDist, List<String> variables) {
            this.jointDist = jointDist;
            this.variables = new ArrayList<>(variables);
            this.random = new Random();
        }

        public List<Map<String, Double>> sample(int numSamples, Map<String, Double> initialState, int burnIn) {
            List<Map<String, Double>> samples = new ArrayList<>();
            Map<String, Double> currentState = new HashMap<>(initialState);

            // 预烧期
            for (int i = 0; i < burnIn; i++) {
                gibbsStep(currentState);
            }

            // 采样期
            for (int i = 0; i < numSamples; i++) {
                gibbsStep(currentState);
                samples.add(new HashMap<>(currentState));
            }

            return samples;
        }

        private void gibbsStep(Map<String, Double> currentState) {
            for (String variable : variables) {
                // 计算条件分布
                ConditionalDistribution condDist = jointDist.getConditional(variable, currentState);

                // 从条件分布中采样
                double newValue = condDist.sample();
                currentState.put(variable, newValue);
            }
        }
    }

    // 接口定义
    public interface ProposalDistribution {
        double sample(double currentState);
        double logDensity(double current, double proposed);
    }

    public interface TargetDistribution {
        double logDensity(double x);
    }

    public interface JointDistribution {
        ConditionalDistribution getConditional(String variable, Map<String, Double> currentAssignments);
    }

    public interface ConditionalDistribution {
        double sample();
        double density(double value);
    }

    // 示例实现：正态分布提议
    public static class NormalProposal implements ProposalDistribution {
        private final double stdDev;

        public NormalProposal(double stdDev) {
            this.stdDev = stdDev;
        }

        @Override
        public double sample(double currentState) {
            return currentState + new Random().nextGaussian() * stdDev;
        }

        @Override
        public double logDensity(double current, double proposed) {
            double diff = proposed - current;
            return -0.5 * (diff * diff) / (stdDev * stdDev) - Math.log(stdDev * Math.sqrt(2 * Math.PI));
        }
    }

    // 测试示例
    public static void main(String[] args) {
        // 示例：采样混合高斯分布
        TargetDistribution mixtureGaussian = x -> {
            double p1 = 0.3 * Math.exp(-0.5 * Math.pow(x - 3, 2)) / Math.sqrt(2 * Math.PI);
            double p2 = 0.7 * Math.exp(-0.5 * Math.pow(x + 2, 2)) / Math.sqrt(2 * Math.PI);
            return Math.log(p1 + p2);
        };

        ProposalDistribution proposal = new NormalProposal(1.0);
        MetropolisHastingsSampler sampler = new MetropolisHastingsSampler(proposal, mixtureGaussian);

        // 采样
        List<Double> samples = sampler.sample(10000, 0.0, 1000);

        // 计算统计量
        double mean = samples.stream().mapToDouble(Double::doubleValue).average().orElse(0);
        double variance = samples.stream()
            .mapToDouble(x -> Math.pow(x - mean, 2))
            .average().orElse(0);

        System.out.printf("样本数量: %d%n", samples.size());
        System.out.printf("均值: %.4f%n", mean);
        System.out.printf("方差: %.4f%n", variance);
    }
}
```
